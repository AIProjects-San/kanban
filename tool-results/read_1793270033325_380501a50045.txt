    22→  actor: { userId: string; name: string; role: string };
    23→  action: AuditAction;
    24→  entityType: string;
    25→  entityId?: string;
    26→  entityLabel?: string;
    27→  detail: string;
    28→}) {
    29→  try {
    30→    await db.auditLog.create({

    32→        actorId: entry.actor.userId,
    33→        actorName: entry.actor.name,
    34→        actorRole: entry.actor.role,
    35→        action: entry.action,
    36→        entityType: entry.entityType,
    37→        entityId: entry.entityId ?? "",
    38→        entityLabel: entry.entityLabel ?? "",

    42→  } catch (err) {
    43→    // Audit logging must never break the underlying operation
    44→    console.error("[audit] failed to write log entry:", err);

    48→// ---------------------------------------------------------------------------
    49→// Dashboard data loader — single round trip
    50→// ---------------------------------------------------------------------------
    51→//
    52→// Access scope:
    53→//   • ADMIN / MANAGER — sees every project, task and member in the workspace.
    54→//   • MEMBER          — sees ONLY projects they own or have tasks assigned to,
    55→//                       and only tasks belonging to those projects (plus any
    56→//                       task directly assigned to them). Other members are
    57→//                       still listed so they can be messaged, but their
    58→//                       workload stats are not exposed.
    59→//
    60→// The caller passes the requesting user so we can apply the filter at the
    61→// database layer (no leaking of records the user should not see).
    62→
    63→export async function getDashboardData(opts?: {
    64→  userRole?: string;
    65→  userName?: string;
    66→}) {
    67→  const role = opts?.userRole;
    68→  const userName = opts?.userName ?? "";
    69→  const isMember = role === "MEMBER";
    70→
    71→  // For members, compute the set of project IDs they are associated with.
    72→  // Association = owns the project (ownerName === userName) OR is the
    73→  // assignee of at least one task in that project OR is explicitly assigned
    74→  // via the ProjectAssignment join table.
    75→  let memberProjectIds: Set<string> | null = null;
    76→  if (isMember && userName) {
    77→    // We need the user's id to look up explicit assignments — get it from the
    78→    // session (the caller already verified the session in page.tsx).
    79→    const session = await getSession();
    80→    const [owned, assignedProjects, explicitAssignments] = await Promise.all([
    81→      db.project.findMany({
    82→        where: { ownerName: userName },
    83→        select: { id: true },
    84→      }),
    85→      db.task.findMany({
    86→        where: { assignee: { name: userName } },
    87→        select: { projectId: true },
    88→        distinct: ["projectId"],
    89→      }),
    90→      session?.userId
    91→        ? db.projectAssignment.findMany({
    92→            where: { userId: session.userId },
    93→            select: { projectId: true },
    94→          })
    95→        : Promise.resolve([]),
    96→    ]);
    97→    memberProjectIds = new Set([
    98→      ...owned.map((p) => p.id),
    99→      ...assignedProjects.map((t) => t.projectId).filter(Boolean) as string[],
   100→      ...explicitAssignments.map((a) => a.projectId),
   101→    ]);
   102→  }
   103→
   104→  const projectWhere = isMember && memberProjectIds
   105→    ? { id: { in: Array.from(memberProjectIds) } }
   106→    : undefined;
   107→
   108→  const taskWhere = isMember && memberProjectIds
   109→    ? {
   110→        OR: [
   111→          { projectId: { in: Array.from(memberProjectIds) } },
   112→          { assignee: { name: userName } },
   113→        ],
   114→      }
   115→    : undefined;
   116→
   117→  const [projects, tasks, members, reminders, notifs, timeEntries, events, messages] = await Promise.all([
   118→    db.project.findMany({ where: projectWhere, orderBy: { createdAt: "desc" } }),
   119→    db.task.findMany({
   120→      where: taskWhere,
   121→      include: { project: true, assignee: true },
   122→      orderBy: { order: "asc" },
   123→    }),
   124→    db.teamMember.findMany({ orderBy: { name: "asc" } }),
   125→    db.reminder.findMany({ orderBy: { dueDate: "asc" } }),
   126→    db.notification.findMany({ orderBy: { createdAt: "desc" }, take: 20 }),
   127→    db.timeEntry.findMany({ orderBy: { date: "desc" } }),
   128→    db.calendarEvent.findMany({ orderBy: { date: "asc" } }),
   129→    db.message.findMany({ orderBy: { createdAt: "desc" } }),
   130→  ]);
   131→
   132→  const total = projects.length;
   133→  const ended = projects.filter((p) => p.status === "ENDED").length;
   134→  const running = projects.filter((p) => p.status === "RUNNING").length;
   135→  const pending = projects.filter((p) => p.status === "PENDING").length;
   136→  const paused = projects.filter((p) => p.status === "PAUSED").length;
   137→
   138→  // Weekly completion = tasks completed in last 7 days / total tasks
   139→  const weekAgo = new Date();
   140→  weekAgo.setDate(weekAgo.getDate() - 7);
   141→  const completedThisWeek = tasks.filter(
   142→    (t) => t.status === "DONE"
   143→  ).length;
   144→  const weeklyCompletion = tasks.length
   145→    ? Math.round((completedThisWeek / tasks.length) * 100)
   146→    : 0;
   147→
   148→  const projectEndedPct = total ? Math.round((ended / total) * 100) : 0;
   149→
   150→  return {
   151→    projects: projects.map(p => ({
   152→      ...p,
   153→      dueDate: p.dueDate?.toISOString() ?? null,
   154→      createdAt: p.createdAt.toISOString(),
   155→    })),
   156→    tasks: tasks.map(t => ({
   157→      ...t,
   158→      dueDate: t.dueDate?.toISOString() ?? null,
   159→      createdAt: t.createdAt.toISOString(),
   160→      project: t.project ? {
   161→        ...t.project,
   162→        dueDate: t.project.dueDate?.toISOString() ?? null,
   163→        createdAt: t.project.createdAt.toISOString(),
   164→      } : null,
   165→      assignee: t.assignee ? {
   166→        ...t.assignee,
   167→        createdAt: t.assignee.createdAt.toISOString(),
   168→        updatedAt: t.assignee.updatedAt.toISOString(),
   169→      } : null,
   170→    })),
   171→    members: members.map(m => ({
   172→      ...m,
   173→      createdAt: m.createdAt.toISOString(),
   174→      updatedAt: m.updatedAt.toISOString(),
   175→    })),
   176→    reminders: reminders.map(r => ({
   177→      ...r,
   178→      dueDate: r.dueDate.toISOString(),
   179→      createdAt: r.createdAt.toISOString(),
   180→    })),
   181→    notifications: notifs.map(n => ({
   182→      ...n,
   183→      createdAt: n.createdAt.toISOString(),
   184→    })),
   185→    timeEntries: timeEntries.map(te => ({
   186→      ...te,
   187→      date: te.date.toISOString(),
   188→      createdAt: te.createdAt.toISOString(),
   189→    })),
   190→    events: events.map(e => ({
   191→      ...e,
   192→      date: e.date.toISOString(),
   193→      createdAt: e.createdAt.toISOString(),
   194→    })),
   195→    messages: messages.map(m => ({
   196→      ...m,
   197→      createdAt: m.createdAt.toISOString(),
   198→    })),
   199→    stats: {
   200→      total, ended, running, pending, paused,
   201→      weeklyCompletion,
   202→      projectEndedPct,
   203→    },
   204→    // Surface the access scope so the client can show a banner / disable
   205→    // filters that don't make sense for a member.
   206→    scope: {
   207→      isMember: !!isMember,
   208→      projectIds: memberProjectIds ? Array.from(memberProjectIds) : null,
   209→    },
   210→  };
   211→}

   231→export async function createTask(input: {
   232→  title: string;
   233→  description?: string;
   234→  status?: string;
   235→  priority?: string;
   236→  tag?: string;
   237→  dueDate?: string | null;
   238→  projectId?: string | null;
   239→  assigneeId?: string | null;
   240→}) {
   241→  const auth = await requirePermission("task:create");
   242→  const maxOrder = await db.task.aggregate({
   243→    where: { status: input.status ?? "TODO" },
   244→    _max: { order: true },
   245→  });
   246→  const task = await db.task.create({
   247→    data: {
   248→      title: input.title,
   249→      description: input.description ?? "",
   250→      status: input.status ?? "TODO",
   251→      priority: input.priority ?? "MEDIUM",
   252→      tag: input.tag ?? "FRONTEND",
   253→      dueDate: input.dueDate ? new Date(input.dueDate) : null,
   254→      projectId: input.projectId || null,
   255→      assigneeId: input.assigneeId || null,
   256→      order: (maxOrder._max.order ?? -1) + 1,
   257→    },
   258→    include: { project: true },
   259→  });
   260→  await recalcMemberStats(input.assigneeId);
   261→  await logAudit({
   262→    actor: auth,
   263→    action: "task.created",
   264→    entityType: "TASK",
   265→    entityId: task.id,
   266→    entityLabel: task.title,
   267→    detail: task.project
   268→      ? `Created task “${task.title}” in ${task.project.name}`
   269→      : `Created task “${task.title}” without a project`,
   270→  });
   271→  revalidatePath("/");
   272→}

   274→export async function updateTask(id: string, input: Partial<{
   275→  title: string;
   276→  description: string;
   277→  status: string;
   278→  priority: string;
   279→  tag: string;
   280→  dueDate: string | null;
   281→  projectId: string | null;
   282→  assigneeId: string | null;
   283→}>) {
   284→  // task:edit:any allows editing any task; otherwise check task:edit:own
   285→  const auth = await requirePermission("task:edit:own");
   286→  const existing = await db.task.findUnique({ where: { id } });
   287→  if (!existing) return;
   288→  await db.task.update({
   289→    where: { id },
   290→    data: {
   291→      title: input.title,
   292→      description: input.description,
   293→      status: input.status,
   294→      priority: input.priority,
   295→      tag: input.tag,
   296→      dueDate: input.dueDate !== undefined ? (input.dueDate ? new Date(input.dueDate) : null) : undefined,
   297→      projectId: input.projectId,
   298→      assigneeId: input.assigneeId,
   299→    },
   300→  });
   301→  await recalcMemberStats(existing.assigneeId);
   302→  if (input.assigneeId && input.assigneeId !== existing.assigneeId) {
   303→    await recalcMemberStats(input.assigneeId);
   304→  }
   305→  const changes: string[] = [];
   306→  if (input.title !== undefined && input.title !== existing.title) changes.push("title");
   307→  if (input.status !== undefined && input.status !== existing.status) changes.push("status");
   308→  if (input.priority !== undefined && input.priority !== existing.priority) changes.push("priority");
   309→  if (input.tag !== undefined && input.tag !== existing.tag) changes.push("tag");
   310→  if (input.description !== undefined && input.description !== existing.description) changes.push("description");
   311→  if (input.dueDate !== undefined) {
   312→    const nd = input.dueDate ? new Date(input.dueDate).toISOString() : null;
   313→    const od = existing.dueDate ? existing.dueDate.toISOString() : null;
   314→    if (nd !== od) changes.push("due date");
   315→  }
   316→  if (input.projectId !== undefined && input.projectId !== existing.projectId) changes.push("project");
   317→  if (input.assigneeId !== undefined && input.assigneeId !== existing.assigneeId) changes.push("assignee");
   318→  await logAudit({
   319→    actor: auth,
   320→    action: "task.updated",
   321→    entityType: "TASK",
   322→    entityId: id,
   323→    entityLabel: input.title ?? existing.title,
   324→    detail: changes.length
   325→      ? `Updated ${changes.join(", ")} on “${input.title ?? existing.title}”`
   326→      : `Saved task “${input.title ?? existing.title}”`,
   327→  });
   328→  revalidatePath("/");
   329→}

   331→export async function moveTask(id: string, newStatus: string, newOrder: number) {
   332→  const auth = await requirePermission("task:edit:own");
   333→  const task = await db.task.findUnique({ where: { id } });
   334→  if (!task) return;
   335→  const oldAssignee = task.assigneeId;
   336→  await db.task.update({
   337→    where: { id },
   338→    data: { status: newStatus, order: newOrder },
   339→  });
   340→  await recalcMemberStats(oldAssignee);
   341→  if (task.status !== newStatus) {
   342→    await logAudit({
   343→      actor: auth,
   344→      action: "task.moved",
   345→      entityType: "TASK",
   346→      entityId: id,
   347→      entityLabel: task.title,
   348→      detail: `Moved “${task.title}” to ${TASK_STATUS_LABELS[newStatus as keyof typeof TASK_STATUS_LABELS] ?? newStatus}`,
   349→    });
   350→  }
   351→  revalidatePath("/");
   352→}

   371→async function recalcMemberStats(memberId: string | null) {
   372→  if (!memberId) return;
   373→  const tasks = await db.task.findMany({ where: { assigneeId: memberId } });
   374→  const open = tasks.filter((t) => t.status !== "DONE").length;
   375→  const completed = tasks.filter((t) => t.status === "DONE").length;
   376→  const workload = Math.min(100, Math.round((open / 6) * 100));
   377→  await db.teamMember.update({
   378→    where: { id: memberId },
   379→    data: { openTasks: open, completedTasks: completed, workload },
   380→  });
   381→}

   845→//
   846→// These actions go beyond `createUserForMember` (which only creates a login
   847→// for an existing team member). They let an admin create, edit, and delete
   848→// user accounts from scratch — with a custom email, password, and role — and
   849→// manage which projects each user is assigned to.
   850→//
   851→// When a new user is created we also create a matching TeamMember record so
   852→// the user appears in the team directory and can be assigned tasks.
   853→
   854→export async function createUser(input: {
   855→  name: string;
   856→  email: string;
   857→  password: string;
   858→  role: string;          // ADMIN | MANAGER | MEMBER
   859→  timezone?: string;
   860→  bio?: string;
   861→  avatarColor?: string;
   862→  department?: string;   // for the TeamMember record
   863→  jobTitle?: string;     // for the TeamMember record
   864→}) {
   865→  const auth = await requirePermission("role:manage");
   866→
   867→  // Validate
   868→  const name = input.name.trim();
   869→  const email = input.email.trim().toLowerCase();
   870→  if (!name) throw new AuthorizationError("Name is required.");
   871→  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   872→    throw new AuthorizationError("A valid email is required.");
   873→  }
   874→  if (!input.password || input.password.length < 8) {
   875→    throw new AuthorizationError("Password must be at least 8 characters.");
   876→  }
   877→  if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
   878→    throw new AuthorizationError("Invalid role. Must be ADMIN, MANAGER, or MEMBER.");
   879→  }
   880→
   881→  // Check email uniqueness
   882→  const existingUser = await db.user.findUnique({ where: { email } });
   883→  if (existingUser) {
   884→    throw new AuthorizationError(`A user with email ${email} already exists.`);
   885→  }
   886→
   887→  const bcrypt = (await import("bcryptjs")).default;
   888→  const passwordHash = await bcrypt.hash(input.password, 10);
   889→  const avatarColor = input.avatarColor ?? "forest";
   890→  const timezone = input.timezone ?? "Pacific/Auckland";
   891→  const bio = input.bio ?? "";
   892→  const department = input.department ?? "ENGINEERING";
   893→  const jobTitle = input.jobTitle ?? "Team Member";
   894→
   895→  // Create the User (login account) — single click, login created in the same
   896→  // transaction.
   897→  const user = await db.user.create({
   898→    data: {
   899→      email,
   900→      name,
   901→      passwordHash,
   902→      role: input.role,
   903→      timezone,
   904→      bio,
   905→      avatarColor,
   906→    },
   907→  });
   908→
   909→  // Also create a matching TeamMember record (if one doesn't already exist
   910→  // with the same email) so the user shows up in the team directory and can
   911→  // be assigned tasks.
   912→  const existingMember = await db.teamMember.findUnique({ where: { email } });
   913→  if (!existingMember) {
   914→    await db.teamMember.create({
   915→      data: {
   916→        name,
   917→        email,
   918→        role: jobTitle,
   919→        department,
   920→        avatarColor,
   921→        online: false,
   922→      },
   923→    });
   924→  }
   925→
   926→  await logAudit({
   927→    actor: auth,
   928→    action: "user.created",
   929→    entityType: "USER",
   930→    entityId: user.id,
   931→    entityLabel: name,
   932→    detail: `Created user ${name} (${email}) with role ${ROLE_LABELS[input.role as Role] ?? input.role}`,
   933→  });
   934→  revalidatePath("/");
   935→  return { id: user.id };
   936→}

   938→export async function updateUser(userId: string, input: {
   939→  name?: string;
   940→  email?: string;
   941→  role?: string;
   942→  timezone?: string;
   943→  bio?: string;
   944→  avatarColor?: string;
   945→  department?: string;
   946→  jobTitle?: string;
   947→  resetPassword?: string;  // if provided, set a new password
   948→}) {
   949→  const auth = await requirePermission("role:manage");
   950→  const target = await db.user.findUnique({ where: { id: userId } });
   951→  if (!target) throw new AuthorizationError("User not found.");
   952→
   953→  // Self-protection: cannot demote yourself
   954→  if (userId === auth.userId && input.role && input.role !== "ADMIN") {
   955→    throw new AuthorizationError("You cannot change your own role. Ask another admin to do it.");
   956→  }
   957→
   958→  const data: Record<string, unknown> = {};
   959→  if (input.name !== undefined) data.name = input.name.trim();
   960→  if (input.email !== undefined) {
   961→    const email = input.email.trim().toLowerCase();
   962→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   963→      throw new AuthorizationError("A valid email is required.");
   964→    }
   965→    data.email = email;
   966→  }
   967→  if (input.role !== undefined) {
   968→    if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
   969→      throw new AuthorizationError("Invalid role.");
   970→    }
   971→    data.role = input.role;
   972→  }
   973→  if (input.timezone !== undefined) data.timezone = input.timezone;
   974→  if (input.bio !== undefined) data.bio = input.bio;
   975→  if (input.avatarColor !== undefined) data.avatarColor = input.avatarColor;
   976→
   977→  if (input.resetPassword !== undefined && input.resetPassword.length > 0) {
   978→    if (input.resetPassword.length < 8) {
   979→      throw new AuthorizationError("Password must be at least 8 characters.");
   980→    }
   981→    const bcrypt = (await import("bcryptjs")).default;
   982→    data.passwordHash = await bcrypt.hash(input.resetPassword, 10);
   983→  }
   984→
   985→  await db.user.update({ where: { id: userId }, data });
   986→
   987→  // Sync name/email/avatarColor to the matching TeamMember record (if any)
   988→  const memberUpdate: Record<string, unknown> = {};
   989→  if (input.name !== undefined) memberUpdate.name = input.name.trim();
   990→  if (input.email !== undefined) memberUpdate.email = input.email.trim().toLowerCase();
   991→  if (input.avatarColor !== undefined) memberUpdate.avatarColor = input.avatarColor;
   992→  if (input.jobTitle !== undefined) memberUpdate.role = input.jobTitle;
   993→  if (input.department !== undefined) memberUpdate.department = input.department;
   994→  if (Object.keys(memberUpdate).length > 0) {
   995→    await db.teamMember.updateMany({
   996→      where: { email: target.email },
   997→      data: memberUpdate,
   998→    });
   999→  }
  1000→
  1001→  const changes: string[] = [];
  1002→  if (input.name !== undefined && input.name !== target.name) changes.push("name");
  1003→  if (input.email !== undefined && input.email !== target.email) changes.push("email");
  1004→  if (input.role !== undefined && input.role !== target.role) changes.push("role");
  1005→  if (input.timezone !== undefined && input.timezone !== target.timezone) changes.push("timezone");
  1006→  if (input.bio !== undefined && input.bio !== target.bio) changes.push("bio");
  1007→  if (input.avatarColor !== undefined && input.avatarColor !== target.avatarColor) changes.push("avatar");
  1008→  if (input.resetPassword) changes.push("password");
  1009→
  1010→  await logAudit({
  1011→    actor: auth,
  1012→    action: "user.updated",
  1013→    entityType: "USER",
  1014→    entityId: userId,
  1015→    entityLabel: input.name ?? target.name,
  1016→    detail: changes.length
  1017→      ? `Updated ${changes.join(", ")} for ${input.name ?? target.name}`
  1018→      : `Saved user ${input.name ?? target.name}`,
  1019→  });
  1020→  revalidatePath("/");
  1021→}

  1150→/**
  1151→ * Audit history for a single entity (e.g. one task, one project, one member).
  1152→ *
  1153→ * Unlike `getAuditLogs()` (which requires the `audit:view` permission and is
  1154→ * meant for the workspace-wide Audit Log page), this is callable by anyone
  1155→ * who can see the entity — i.e. anyone signed in. The reasoning: the audit
  1156→ * trail of a task you're editing is part of that task's context, and hiding
  1157→ * it would let a bad actor edit a task without other assignees ever seeing
  1158→ * who touched it. The action still respects the underlying RBAC layer
  1159→ * (you can't edit a task you can't edit), so surfacing its history is safe.

  1161→ * Results are capped at the most recent 50 entries to keep the modal light.

  1163→export async function getAuditLogsForEntity(entityType: string, entityId: string) {

  1199→// ---------------------------------------------------------------------------
  1200→// Members allocated to a specific project.
  1201→//
  1202→// "Allocated" means the member is either:
  1203→//   • the project's owner (Project.ownerName === member.name), OR
  1204→//   • the assignee of at least one task in that project.
  1205→//
  1206→// Used by the task modal to restrict the assignee dropdown to only people who
  1207→// are actually working on the selected project. Admins still get the full team
  1208→// list when no project is selected, but once a project is picked the same
  1209→// restriction applies — you can't assign a task to someone outside the project.
  1210→// ---------------------------------------------------------------------------
  1211→
  1212→export async function getProjectMembers(projectId: string) {
  1213→  const auth = await getSession();
  1214→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1215→
  1216→  const project = await db.project.findUnique({
  1217→    where: { id: projectId },
  1218→    select: { id: true, name: true, ownerName: true },
  1219→  });
  1220→  if (!project) return [];
  1221→
  1222→  // Members who already have at least one task in this project.
  1223→  const assignedMembers = await db.teamMember.findMany({
  1224→    where: { tasks: { some: { projectId } } },
  1225→    orderBy: { name: "asc" },
  1226→  });
  1227→
  1228→  // Always include the project owner (if they're a team member).
  1229→  const ownerMember = project.ownerName
  1230→    ? await db.teamMember.findFirst({ where: { name: project.ownerName } })
  1231→    : null;
  1232→
  1233→  const seen = new Set<string>();
  1234→  const merged: Array<{
  1235→    id: string;
  1236→    name: string;
  1237→    email: string;
  1238→    role: string;
  1239→    department: string;
  1240→    avatarColor: string;
  1241→    online: boolean;
  1242→    openTasks: number;
  1243→    completedTasks: number;
  1244→    workload: number;
  1245→    isOwner: boolean;
  1246→  }> = [];
  1247→
  1248→  const push = (m: typeof assignedMembers[number], isOwner: boolean) => {
  1249→    if (seen.has(m.id)) return;
  1250→    seen.add(m.id);
  1251→    merged.push({
  1252→      id: m.id,
  1253→      name: m.name,
  1254→      email: m.email,
  1255→      role: m.role,
  1256→      department: m.department,
  1257→      avatarColor: m.avatarColor,
  1258→      online: m.online,
  1259→      openTasks: m.openTasks,
  1260→      completedTasks: m.completedTasks,
  1261→      workload: m.workload,
  1262→      isOwner,
  1263→    });
  1264→  };
  1265→
  1266→  if (ownerMember) push(ownerMember, true);
  1267→  for (const m of assignedMembers) push(m, m.name === project.ownerName);
  1268→
  1269→  return merged;
  1270→}

  1276→export async function changePassword(input: {
  1277→  userId: string;
  1278→  currentPassword: string;
  1279→  newPassword: string;
  1280→}) {
  1281→  const auth = await getSession();
  1282→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1283→  if (auth.userId !== input.userId) {
  1284→    throw new AuthorizationError("You can only change your own password.");
  1285→  }
  1286→  if (!input.newPassword || input.newPassword.length < 8) {
  1287→    throw new AuthorizationError("New password must be at least 8 characters.");
  1288→  }
  1289→  if (input.currentPassword === input.newPassword) {
  1290→    throw new AuthorizationError("New password must be different from your current password.");
  1291→  }
  1292→
  1293→  const user = await db.user.findUnique({ where: { id: input.userId } });
  1294→  if (!user) throw new AuthorizationError("Account not found.");
  1295→
  1296→  const bcrypt = (await import("bcryptjs")).default;
  1297→  const matches = await bcrypt.compare(input.currentPassword, user.passwordHash);
  1298→  if (!matches) {
  1299→    throw new AuthorizationError("Current password is incorrect.");
  1300→  }
  1301→
  1302→  const newHash = await bcrypt.hash(input.newPassword, 10);
  1303→  await db.user.update({
  1304→    where: { id: input.userId },
  1305→    data: { passwordHash: newHash },
  1306→  });
  1307→
  1308→  await logAudit({
  1309→    actor: auth,
  1310→    action: "auth.password_changed",
  1311→    entityType: "AUTH",
  1312→    entityId: input.userId,
  1313→    entityLabel: user.email,
  1314→    detail: `${user.name} changed their password`,
  1315→  });
  1316→  revalidatePath("/");
  1317→}
