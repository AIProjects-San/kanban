    57→    57→export async function getDashboardData() {
    58→    58→  const [projects, tasks, members, reminders, notifs, timeEntries, events, messages] = await Promise.all([
    59→    59→    db.project.findMany({ orderBy: { createdAt: "desc" } }),
    60→    60→    db.task.findMany({ include: { project: true, assignee: true }, orderBy: { order: "asc" } }),
    61→    61→    db.teamMember.findMany({ orderBy: { name: "asc" } }),
    62→    62→    db.reminder.findMany({ orderBy: { dueDate: "asc" } }),
    63→    63→    db.notification.findMany({ orderBy: { createdAt: "desc" }, take: 20 }),
    64→    64→    db.timeEntry.findMany({ orderBy: { date: "desc" } }),
    65→    65→    db.calendarEvent.findMany({ orderBy: { date: "asc" } }),
    66→    66→    db.message.findMany({ orderBy: { createdAt: "desc" } }),
    67→    67→  ]);
    68→    68→
    69→    69→  const total = projects.length;
    70→    70→  const ended = projects.filter((p) => p.status === "ENDED").length;
    71→    71→  const running = projects.filter((p) => p.status === "RUNNING").length;
    72→    72→  const pending = projects.filter((p) => p.status === "PENDING").length;
    73→    73→  const paused = projects.filter((p) => p.status === "PAUSED").length;
    74→    74→
    75→    75→  // Weekly completion = tasks completed in last 7 days / total tasks
    76→    76→  const weekAgo = new Date();
    77→    77→  weekAgo.setDate(weekAgo.getDate() - 7);
    78→    78→  const completedThisWeek = tasks.filter(
    79→    79→    (t) => t.status === "DONE"
    80→    80→  ).length;
    81→    81→  const weeklyCompletion = tasks.length
    82→    82→    ? Math.round((completedThisWeek / tasks.length) * 100)
    83→    83→    : 0;
    84→    84→
    85→    85→  const projectEndedPct = total ? Math.round((ended / total) * 100) : 0;
    86→    86→
    87→    87→  return {
    88→    88→    projects: projects.map(p => ({
    89→    89→      ...p,
    90→    90→      dueDate: p.dueDate?.toISOString() ?? null,
    91→    91→      createdAt: p.createdAt.toISOString(),
    92→    92→    })),
    93→    93→    tasks: tasks.map(t => ({
    94→    94→      ...t,
    95→    95→      dueDate: t.dueDate?.toISOString() ?? null,
    96→    96→      createdAt: t.createdAt.toISOString(),
    97→    97→      project: t.project ? {
    98→    98→        ...t.project,
    99→    99→        dueDate: t.project.dueDate?.toISOString() ?? null,
   100→   100→        createdAt: t.project.createdAt.toISOString(),
   101→   101→      } : null,
   102→   102→      assignee: t.assignee ? {
   103→   103→        ...t.assignee,
   104→   104→        createdAt: t.assignee.createdAt.toISOString(),
   105→   105→        updatedAt: t.assignee.updatedAt.toISOString(),
   106→   106→      } : null,
   107→   107→    })),
   108→   108→    members: members.map(m => ({
   109→   109→      ...m,
   110→   110→      createdAt: m.createdAt.toISOString(),
   111→   111→      updatedAt: m.updatedAt.toISOString(),
   112→   112→    })),
   113→   113→    reminders: reminders.map(r => ({
   114→   114→      ...r,
   115→   115→      dueDate: r.dueDate.toISOString(),
   116→   116→      createdAt: r.createdAt.toISOString(),
   117→   117→    })),
   118→   118→    notifications: notifs.map(n => ({
   119→   119→      ...n,
   120→   120→      createdAt: n.createdAt.toISOString(),
   121→   121→    })),
   122→   122→    timeEntries: timeEntries.map(te => ({
   123→   123→      ...te,
   124→   124→      date: te.date.toISOString(),
   125→   125→      createdAt: te.createdAt.toISOString(),
   126→   126→    })),
   127→   127→    events: events.map(e => ({
   128→   128→      ...e,
   129→   129→      date: e.date.toISOString(),
   130→   130→      createdAt: e.createdAt.toISOString(),
   131→   131→    })),
   132→   132→    messages: messages.map(m => ({
   133→   133→      ...m,
   134→   134→      createdAt: m.createdAt.toISOString(),
   135→   135→    })),
   136→   136→    stats: {
   137→   137→      total, ended, running, pending, paused,
   138→   138→      weeklyCompletion,
   139→   139→      projectEndedPct,
   140→   140→    },
   141→   141→  };
   142→   142→}

   162→   162→export async function createTask(input: {
   163→   163→  title: string;
   164→   164→  description?: string;
   165→   165→  status?: string;
   166→   166→  priority?: string;
   167→   167→  tag?: string;
   168→   168→  dueDate?: string | null;
   169→   169→  projectId?: string | null;
   170→   170→  assigneeId?: string | null;
   171→   171→}) {
   172→   172→  const auth = await requirePermission("task:create");
   173→   173→  const maxOrder = await db.task.aggregate({
   174→   174→    where: { status: input.status ?? "TODO" },
   175→   175→    _max: { order: true },
   176→   176→  });
   177→   177→  const task = await db.task.create({
   178→   178→    data: {
   179→   179→      title: input.title,
   180→   180→      description: input.description ?? "",
   181→   181→      status: input.status ?? "TODO",
   182→   182→      priority: input.priority ?? "MEDIUM",
   183→   183→      tag: input.tag ?? "FRONTEND",
   184→   184→      dueDate: input.dueDate ? new Date(input.dueDate) : null,
   185→   185→      projectId: input.projectId || null,
   186→   186→      assigneeId: input.assigneeId || null,
   187→   187→      order: (maxOrder._max.order ?? -1) + 1,
   188→   188→    },
   189→   189→    include: { project: true },
   190→   190→  });
   191→   191→  await recalcMemberStats(input.assigneeId);
   192→   192→  await logAudit({
   193→   193→    actor: auth,
   194→   194→    action: "task.created",
   195→   195→    entityType: "TASK",
   196→   196→    entityId: task.id,
   197→   197→    entityLabel: task.title,
   198→   198→    detail: task.project
   199→   199→      ? `Created task “${task.title}” in ${task.project.name}`
   200→   200→      : `Created task “${task.title}” without a project`,
   201→   201→  });
   202→   202→  revalidatePath("/");

   205→   205→export async function updateTask(id: string, input: Partial<{
   206→   206→  title: string;
   207→   207→  description: string;
   208→   208→  status: string;
   209→   209→  priority: string;
   210→   210→  tag: string;
   211→   211→  dueDate: string | null;
   212→   212→  projectId: string | null;
   213→   213→  assigneeId: string | null;
   214→   214→}>) {
   215→   215→  // task:edit:any allows editing any task; otherwise check task:edit:own
   216→   216→  const auth = await requirePermission("task:edit:own");
   217→   217→  const existing = await db.task.findUnique({ where: { id } });
   218→   218→  if (!existing) return;
   219→   219→  await db.task.update({
   220→   220→    where: { id },
   221→   221→    data: {
   222→   222→      title: input.title,
   223→   223→      description: input.description,
   224→   224→      status: input.status,
   225→   225→      priority: input.priority,
   226→   226→      tag: input.tag,
   227→   227→      dueDate: input.dueDate !== undefined ? (input.dueDate ? new Date(input.dueDate) : null) : undefined,
   228→   228→      projectId: input.projectId,
   229→   229→      assigneeId: input.assigneeId,
   230→   230→    },
   231→   231→  });
   232→   232→  await recalcMemberStats(existing.assigneeId);
   233→   233→  if (input.assigneeId && input.assigneeId !== existing.assigneeId) {
   234→   234→    await recalcMemberStats(input.assigneeId);
   235→   235→  }
   236→   236→  const changes: string[] = [];
   237→   237→  if (input.title !== undefined && input.title !== existing.title) changes.push("title");
   238→   238→  if (input.status !== undefined && input.status !== existing.status) changes.push("status");
   239→   239→  if (input.priority !== undefined && input.priority !== existing.priority) changes.push("priority");
   240→   240→  if (input.tag !== undefined && input.tag !== existing.tag) changes.push("tag");
   241→   241→  if (input.description !== undefined && input.description !== existing.description) changes.push("description");
   242→   242→  if (input.dueDate !== undefined) {
   243→   243→    const nd = input.dueDate ? new Date(input.dueDate).toISOString() : null;
   244→   244→    const od = existing.dueDate ? existing.dueDate.toISOString() : null;
   245→   245→    if (nd !== od) changes.push("due date");
   246→   246→  }
   247→   247→  if (input.projectId !== undefined && input.projectId !== existing.projectId) changes.push("project");
   248→   248→  if (input.assigneeId !== undefined && input.assigneeId !== existing.assigneeId) changes.push("assignee");
   249→   249→  await logAudit({
   250→   250→    actor: auth,
   251→   251→    action: "task.updated",
   252→   252→    entityType: "TASK",
   253→   253→    entityId: id,
   254→   254→    entityLabel: input.title ?? existing.title,
   255→   255→    detail: changes.length
   256→   256→      ? `Updated ${changes.join(", ")} on “${input.title ?? existing.title}”`
   257→   257→      : `Saved task “${input.title ?? existing.title}”`,
   258→   258→  });
   259→   259→  revalidatePath("/");
   260→   260→}

   262→   262→export async function moveTask(id: string, newStatus: string, newOrder: number) {
   263→   263→  const auth = await requirePermission("task:edit:own");
   264→   264→  const task = await db.task.findUnique({ where: { id } });
   265→   265→  if (!task) return;
   266→   266→  const oldAssignee = task.assigneeId;
   267→   267→  await db.task.update({
   268→   268→    where: { id },
   269→   269→    data: { status: newStatus, order: newOrder },
   270→   270→  });
   271→   271→  await recalcMemberStats(oldAssignee);
   272→   272→  if (task.status !== newStatus) {
   273→   273→    await logAudit({
   274→   274→      actor: auth,
   275→   275→      action: "task.moved",
   276→   276→      entityType: "TASK",
   277→   277→      entityId: id,
   278→   278→      entityLabel: task.title,
   279→   279→      detail: `Moved “${task.title}” to ${STATUS_LABELS[newStatus] ?? newStatus}`,
   280→   280→    });
   281→   281→  }
   282→   282→  revalidatePath("/");
   283→   283→}

   302→   302→async function recalcMemberStats(memberId: string | null) {
   303→   303→  if (!memberId) return;

   305→   305→  const open = tasks.filter((t) => t.status !== "DONE").length;
   306→   306→  const completed = tasks.filter((t) => t.status === "DONE").length;
   307→   307→  const workload = Math.min(100, Math.round((open / 6) * 100));
   308→   308→  await db.teamMember.update({
   309→   309→    where: { id: memberId },
   310→   310→    data: { openTasks: open, completedTasks: completed, workload },

   632→   632→    if (input.bio !== existing.bio) changes.push("bio");

   634→   634→    await logAudit({
   635→   635→      actor: auth,
   636→   636→      action: "profile.updated",
   637→   637→      entityType: "USER",
   638→   638→      entityId: userId,
   639→   639→      entityLabel: input.name,
   640→   640→      detail: changes.length
   641→   641→        ? `Updated ${changes.join(", ")} on their profile`
   642→   642→        : "Saved their profile",
   643→   643→    });
   644→   644→  }
