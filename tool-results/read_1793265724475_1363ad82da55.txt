    56→    56→//
    57→    57→// Access scope:
    58→    58→//   • ADMIN / MANAGER — sees every project, task and member in the workspace.
    59→    59→//   • MEMBER          — sees ONLY projects they own or have tasks assigned to,
    60→    60→//                       and only tasks belonging to those projects (plus any
    61→    61→//                       task directly assigned to them). Other members are
    62→    62→//                       still listed so they can be messaged, but their
    63→    63→//                       workload stats are not exposed.
    64→    64→//
    65→    65→// The caller passes the requesting user so we can apply the filter at the
    66→    66→// database layer (no leaking of records the user should not see).

    68→    68→export async function getDashboardData(opts?: {
    69→    69→  userRole?: string;
    70→    70→  userName?: string;
    71→    71→}) {
    72→    72→  const role = opts?.userRole;
    73→    73→  const userName = opts?.userName ?? "";
    74→    74→  const isMember = role === "MEMBER";
    75→    75→
    76→    76→  // For members, compute the set of project IDs they are associated with.
    77→    77→  // Association = owns the project (ownerName === userName) OR is the
    78→    78→  // assignee of at least one task in that project OR is explicitly assigned
    79→    79→  // via the ProjectAssignment join table.
    80→    80→  let memberProjectIds: Set<string> | null = null;
    81→    81→  if (isMember && userName) {
    82→    82→    // We need the user's id to look up explicit assignments — get it from the
    83→    83→    // session (the caller already verified the session in page.tsx).
    84→    84→    const session = await getSession();
    85→    85→    const [owned, assignedProjects, explicitAssignments] = await Promise.all([
    86→    86→      db.project.findMany({
    87→    87→        where: { ownerName: userName },
    88→    88→        select: { id: true },
    89→    89→      }),
    90→    90→      db.task.findMany({
    91→    91→        where: { assignee: { name: userName } },
    92→    92→        select: { projectId: true },
    93→    93→        distinct: ["projectId"],
    94→    94→      }),
    95→    95→      session?.userId
    96→    96→        ? db.projectAssignment.findMany({
    97→    97→            where: { userId: session.userId },
    98→    98→            select: { projectId: true },
    99→    99→          })
   100→   100→        : Promise.resolve([]),
   101→   101→    ]);
   102→   102→    memberProjectIds = new Set([
   103→   103→      ...owned.map((p) => p.id),
   104→   104→      ...assignedProjects.map((t) => t.projectId).filter(Boolean) as string[],
   105→   105→      ...explicitAssignments.map((a) => a.projectId),
   106→   106→    ]);
   107→   107→  }
   108→   108→
   109→   109→  const projectWhere = isMember && memberProjectIds
   110→   110→    ? { id: { in: Array.from(memberProjectIds) } }
   111→   111→    : undefined;
   112→   112→
   113→   113→  const taskWhere = isMember && memberProjectIds
   114→   114→    ? {
   115→   115→        OR: [
   116→   116→          { projectId: { in: Array.from(memberProjectIds) } },
   117→   117→          { assignee: { name: userName } },
   118→   118→        ],
   119→   119→      }
   120→   120→    : undefined;
   121→   121→
   122→   122→  const [projects, tasks, members, reminders, notifs, timeEntries, events, messages] = await Promise.all([
   123→   123→    db.project.findMany({ where: projectWhere, orderBy: { createdAt: "desc" } }),
   124→   124→    db.task.findMany({
   125→   125→      where: taskWhere,
   126→   126→      include: { project: true, assignee: true },
   127→   127→      orderBy: { order: "asc" },
   128→   128→    }),
   129→   129→    db.teamMember.findMany({ orderBy: { name: "asc" } }),
   130→   130→    db.reminder.findMany({ orderBy: { dueDate: "asc" } }),
   131→   131→    db.notification.findMany({ orderBy: { createdAt: "desc" }, take: 20 }),
   132→   132→    db.timeEntry.findMany({ orderBy: { date: "desc" } }),
   133→   133→    db.calendarEvent.findMany({ orderBy: { date: "asc" } }),
   134→   134→    db.message.findMany({ orderBy: { createdAt: "desc" } }),
   135→   135→  ]);
   136→   136→
   137→   137→  const total = projects.length;
   138→   138→  const ended = projects.filter((p) => p.status === "ENDED").length;
   139→   139→  const running = projects.filter((p) => p.status === "RUNNING").length;
   140→   140→  const pending = projects.filter((p) => p.status === "PENDING").length;
   141→   141→  const paused = projects.filter((p) => p.status === "PAUSED").length;
   142→   142→
   143→   143→  // Weekly completion = tasks completed in last 7 days / total tasks
   144→   144→  const weekAgo = new Date();
   145→   145→  weekAgo.setDate(weekAgo.getDate() - 7);
   146→   146→  const completedThisWeek = tasks.filter(
   147→   147→    (t) => t.status === "DONE"
   148→   148→  ).length;
   149→   149→  const weeklyCompletion = tasks.length
   150→   150→    ? Math.round((completedThisWeek / tasks.length) * 100)
   151→   151→    : 0;
   152→   152→
   153→   153→  const projectEndedPct = total ? Math.round((ended / total) * 100) : 0;
   154→   154→
   155→   155→  return {
   156→   156→    projects: projects.map(p => ({
   157→   157→      ...p,
   158→   158→      dueDate: p.dueDate?.toISOString() ?? null,
   159→   159→      createdAt: p.createdAt.toISOString(),
   160→   160→    })),
   161→   161→    tasks: tasks.map(t => ({
   162→   162→      ...t,
   163→   163→      dueDate: t.dueDate?.toISOString() ?? null,
   164→   164→      createdAt: t.createdAt.toISOString(),
   165→   165→      project: t.project ? {
   166→   166→        ...t.project,
   167→   167→        dueDate: t.project.dueDate?.toISOString() ?? null,
   168→   168→        createdAt: t.project.createdAt.toISOString(),
   169→   169→      } : null,
   170→   170→      assignee: t.assignee ? {
   171→   171→        ...t.assignee,
   172→   172→        createdAt: t.assignee.createdAt.toISOString(),
   173→   173→        updatedAt: t.assignee.updatedAt.toISOString(),
   174→   174→      } : null,
   175→   175→    })),
   176→   176→    members: members.map(m => ({
   177→   177→      ...m,
   178→   178→      createdAt: m.createdAt.toISOString(),
   179→   179→      updatedAt: m.updatedAt.toISOString(),
   180→   180→    })),
   181→   181→    reminders: reminders.map(r => ({
   182→   182→      ...r,
   183→   183→      dueDate: r.dueDate.toISOString(),
   184→   184→      createdAt: r.createdAt.toISOString(),
   185→   185→    })),
   186→   186→    notifications: notifs.map(n => ({
   187→   187→      ...n,
   188→   188→      createdAt: n.createdAt.toISOString(),
   189→   189→    })),
   190→   190→    timeEntries: timeEntries.map(te => ({
   191→   191→      ...te,
   192→   192→      date: te.date.toISOString(),
   193→   193→      createdAt: te.createdAt.toISOString(),
   194→   194→    })),
   195→   195→    events: events.map(e => ({
   196→   196→      ...e,
   197→   197→      date: e.date.toISOString(),
   198→   198→      createdAt: e.createdAt.toISOString(),
   199→   199→    })),
   200→   200→    messages: messages.map(m => ({
   201→   201→      ...m,
   202→   202→      createdAt: m.createdAt.toISOString(),
   203→   203→    })),
   204→   204→    stats: {
   205→   205→      total, ended, running, pending, paused,
   206→   206→      weeklyCompletion,
   207→   207→      projectEndedPct,
   208→   208→    },
   209→   209→    // Surface the access scope so the client can show a banner / disable
   210→   210→    // filters that don't make sense for a member.
   211→   211→    scope: {
   212→   212→      isMember: !!isMember,
   213→   213→      projectIds: memberProjectIds ? Array.from(memberProjectIds) : null,
   214→   214→    },
   215→   215→  };
   216→   216→}

   236→   236→export async function createTask(input: {
   237→   237→  title: string;
   238→   238→  description?: string;
   239→   239→  status?: string;
   240→   240→  priority?: string;
   241→   241→  tag?: string;
   242→   242→  dueDate?: string | null;
   243→   243→  projectId?: string | null;
   244→   244→  assigneeId?: string | null;
   245→   245→}) {
   246→   246→  const auth = await requirePermission("task:create");
   247→   247→  const maxOrder = await db.task.aggregate({
   248→   248→    where: { status: input.status ?? "TODO" },
   249→   249→    _max: { order: true },
   250→   250→  });
   251→   251→  const task = await db.task.create({
   252→   252→    data: {
   253→   253→      title: input.title,
   254→   254→      description: input.description ?? "",
   255→   255→      status: input.status ?? "TODO",
   256→   256→      priority: input.priority ?? "MEDIUM",
   257→   257→      tag: input.tag ?? "FRONTEND",
   258→   258→      dueDate: input.dueDate ? new Date(input.dueDate) : null,
   259→   259→      projectId: input.projectId || null,
   260→   260→      assigneeId: input.assigneeId || null,
   261→   261→      order: (maxOrder._max.order ?? -1) + 1,
   262→   262→    },
   263→   263→    include: { project: true },
   264→   264→  });
   265→   265→  await recalcMemberStats(input.assigneeId);
   266→   266→  await logAudit({
   267→   267→    actor: auth,
   268→   268→    action: "task.created",
   269→   269→    entityType: "TASK",
   270→   270→    entityId: task.id,
   271→   271→    entityLabel: task.title,
   272→   272→    detail: task.project
   273→   273→      ? `Created task “${task.title}” in ${task.project.name}`
   274→   274→      : `Created task “${task.title}” without a project`,
   275→   275→  });
   276→   276→  revalidatePath("/");
   277→   277→}

   279→   279→export async function updateTask(id: string, input: Partial<{
   280→   280→  title: string;
   281→   281→  description: string;
   282→   282→  status: string;
   283→   283→  priority: string;
   284→   284→  tag: string;
   285→   285→  dueDate: string | null;
   286→   286→  projectId: string | null;
   287→   287→  assigneeId: string | null;
   288→   288→}>) {
   289→   289→  // task:edit:any allows editing any task; otherwise check task:edit:own
   290→   290→  const auth = await requirePermission("task:edit:own");
   291→   291→  const existing = await db.task.findUnique({ where: { id } });
   292→   292→  if (!existing) return;
   293→   293→  await db.task.update({
   294→   294→    where: { id },
   295→   295→    data: {
   296→   296→      title: input.title,
   297→   297→      description: input.description,
   298→   298→      status: input.status,
   299→   299→      priority: input.priority,
   300→   300→      tag: input.tag,
   301→   301→      dueDate: input.dueDate !== undefined ? (input.dueDate ? new Date(input.dueDate) : null) : undefined,
   302→   302→      projectId: input.projectId,
   303→   303→      assigneeId: input.assigneeId,
   304→   304→    },
   305→   305→  });
   306→   306→  await recalcMemberStats(existing.assigneeId);
   307→   307→  if (input.assigneeId && input.assigneeId !== existing.assigneeId) {
   308→   308→    await recalcMemberStats(input.assigneeId);
   309→   309→  }
   310→   310→  const changes: string[] = [];
   311→   311→  if (input.title !== undefined && input.title !== existing.title) changes.push("title");
   312→   312→  if (input.status !== undefined && input.status !== existing.status) changes.push("status");
   313→   313→  if (input.priority !== undefined && input.priority !== existing.priority) changes.push("priority");
   314→   314→  if (input.tag !== undefined && input.tag !== existing.tag) changes.push("tag");
   315→   315→  if (input.description !== undefined && input.description !== existing.description) changes.push("description");
   316→   316→  if (input.dueDate !== undefined) {
   317→   317→    const nd = input.dueDate ? new Date(input.dueDate).toISOString() : null;
   318→   318→    const od = existing.dueDate ? existing.dueDate.toISOString() : null;
   319→   319→    if (nd !== od) changes.push("due date");
   320→   320→  }
   321→   321→  if (input.projectId !== undefined && input.projectId !== existing.projectId) changes.push("project");
   322→   322→  if (input.assigneeId !== undefined && input.assigneeId !== existing.assigneeId) changes.push("assignee");
   323→   323→  await logAudit({
   324→   324→    actor: auth,
   325→   325→    action: "task.updated",
   326→   326→    entityType: "TASK",
   327→   327→    entityId: id,
   328→   328→    entityLabel: input.title ?? existing.title,
   329→   329→    detail: changes.length
   330→   330→      ? `Updated ${changes.join(", ")} on “${input.title ?? existing.title}”`
   331→   331→      : `Saved task “${input.title ?? existing.title}”`,
   332→   332→  });
   333→   333→  revalidatePath("/");
   334→   334→}

   336→   336→export async function moveTask(id: string, newStatus: string, newOrder: number) {
   337→   337→  const auth = await requirePermission("task:edit:own");
   338→   338→  const task = await db.task.findUnique({ where: { id } });
   339→   339→  if (!task) return;
   340→   340→  const oldAssignee = task.assigneeId;
   341→   341→  await db.task.update({
   342→   342→    where: { id },
   343→   343→    data: { status: newStatus, order: newOrder },
   344→   344→  });
   345→   345→  await recalcMemberStats(oldAssignee);
   346→   346→  if (task.status !== newStatus) {
   347→   347→    await logAudit({
   348→   348→      actor: auth,
   349→   349→      action: "task.moved",
   350→   350→      entityType: "TASK",
   351→   351→      entityId: id,
   352→   352→      entityLabel: task.title,
   353→   353→      detail: `Moved “${task.title}” to ${STATUS_LABELS[newStatus] ?? newStatus}`,
   354→   354→    });
   355→   355→  }
   356→   356→  revalidatePath("/");
   357→   357→}

   376→   376→async function recalcMemberStats(memberId: string | null) {
   377→   377→  if (!memberId) return;
   378→   378→  const tasks = await db.task.findMany({ where: { assigneeId: memberId } });
   379→   379→  const open = tasks.filter((t) => t.status !== "DONE").length;
   380→   380→  const completed = tasks.filter((t) => t.status === "DONE").length;
   381→   381→  const workload = Math.min(100, Math.round((open / 6) * 100));
   382→   382→  await db.teamMember.update({
   383→   383→    where: { id: memberId },
   384→   384→    data: { openTasks: open, completedTasks: completed, workload },
   385→   385→  });
   386→   386→}

   911→   911→//
   912→   912→// These actions go beyond `createUserForMember` (which only creates a login
   913→   913→// for an existing team member). They let an admin create, edit, and delete
   914→   914→// user accounts from scratch — with a custom email, password, and role — and
   915→   915→// manage which projects each user is assigned to.
   916→   916→//
   917→   917→// When a new user is created we also create a matching TeamMember record so
   918→   918→// the user appears in the team directory and can be assigned tasks.
   919→   919→
   920→   920→export async function createUser(input: {
   921→   921→  name: string;
   922→   922→  email: string;
   923→   923→  password: string;
   924→   924→  role: string;          // ADMIN | MANAGER | MEMBER
   925→   925→  timezone?: string;
   926→   926→  bio?: string;
   927→   927→  avatarColor?: string;
   928→   928→  department?: string;   // for the TeamMember record
   929→   929→  jobTitle?: string;     // for the TeamMember record
   930→   930→}) {
   931→   931→  const auth = await requirePermission("role:manage");
   932→   932→
   933→   933→  // Validate
   934→   934→  const name = input.name.trim();
   935→   935→  const email = input.email.trim().toLowerCase();
   936→   936→  if (!name) throw new AuthorizationError("Name is required.");
   937→   937→  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   938→   938→    throw new AuthorizationError("A valid email is required.");
   939→   939→  }
   940→   940→  if (!input.password || input.password.length < 8) {
   941→   941→    throw new AuthorizationError("Password must be at least 8 characters.");
   942→   942→  }
   943→   943→  if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
   944→   944→    throw new AuthorizationError("Invalid role. Must be ADMIN, MANAGER, or MEMBER.");
   945→   945→  }
   946→   946→
   947→   947→  // Check email uniqueness
   948→   948→  const existingUser = await db.user.findUnique({ where: { email } });
   949→   949→  if (existingUser) {
   950→   950→    throw new AuthorizationError(`A user with email ${email} already exists.`);
   951→   951→  }
   952→   952→
   953→   953→  const bcrypt = (await import("bcryptjs")).default;
   954→   954→  const passwordHash = await bcrypt.hash(input.password, 10);
   955→   955→  const avatarColor = input.avatarColor ?? "forest";
   956→   956→  const timezone = input.timezone ?? "Pacific/Auckland";
   957→   957→  const bio = input.bio ?? "";
   958→   958→  const department = input.department ?? "ENGINEERING";
   959→   959→  const jobTitle = input.jobTitle ?? "Team Member";
   960→   960→
   961→   961→  // Create the User (login account) — single click, login created in the same
   962→   962→  // transaction.

  1004→  1004→export async function updateUser(userId: string, input: {
  1005→  1005→  name?: string;
  1006→  1006→  email?: string;
  1007→  1007→  role?: string;
  1008→  1008→  timezone?: string;
  1009→  1009→  bio?: string;

  1058→  1058→  if (input.jobTitle !== undefined) memberUpdate.role = input.jobTitle;
  1059→  1059→  if (input.department !== undefined) memberUpdate.department = input.department;

  1062→  1062→      where: { email: target.email },
  1063→  1063→      data: memberUpdate,
  1064→  1064→    });

  1070→  1070→  if (input.role !== undefined && input.role !== target.role) changes.push("role");

  1073→  1073→  if (input.avatarColor !== undefined && input.avatarColor !== target.avatarColor) changes.push("avatar");
  1074→  1074→  if (input.resetPassword) changes.push("password");
  1075→  1075→
  1076→  1076→  await logAudit({
  1077→  1077→    actor: auth,
  1078→  1078→    action: "user.updated",
  1079→  1079→    entityType: "USER",
  1080→  1080→    entityId: userId,
  1081→  1081→    entityLabel: input.name ?? target.name,
  1082→  1082→    detail: changes.length
  1083→  1083→      ? `Updated ${changes.join(", ")} for ${input.name ?? target.name}`
