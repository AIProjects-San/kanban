    26→async function logAudit(entry: {
    27→  actor: { userId: string; name: string; role: string };
    28→  action: string;
    29→  entityType: string;
    30→  entityId?: string;
    31→  entityLabel?: string;
    32→  detail: string;
    33→}) {
    34→  try {
    35→    await db.auditLog.create({
    36→      data: {
    37→        actorId: entry.actor.userId,
    38→        actorName: entry.actor.name,
    39→        actorRole: entry.actor.role,
    40→        action: entry.action,
    41→        entityType: entry.entityType,
    42→        entityId: entry.entityId ?? "",
    43→        entityLabel: entry.entityLabel ?? "",
    44→        detail: entry.detail,
    45→      },
    46→    });
    47→  } catch (err) {
    48→    // Audit logging must never break the underlying operation
    49→    console.error("[audit] failed to write log entry:", err);
    50→  }
    51→}

    57→export async function getDashboardData() {
    58→  const [projects, tasks, members, reminders, notifs, timeEntries, events, messages] = await Promise.all([
    59→    db.project.findMany({ orderBy: { createdAt: "desc" } }),
    60→    db.task.findMany({ include: { project: true, assignee: true }, orderBy: { order: "asc" } }),
    61→    db.teamMember.findMany({ orderBy: { name: "asc" } }),
    62→    db.reminder.findMany({ orderBy: { dueDate: "asc" } }),
    63→    db.notification.findMany({ orderBy: { createdAt: "desc" }, take: 20 }),
    64→    db.timeEntry.findMany({ orderBy: { date: "desc" } }),
    65→    db.calendarEvent.findMany({ orderBy: { date: "asc" } }),
    66→    db.message.findMany({ orderBy: { createdAt: "desc" } }),
    67→  ]);
    68→
    69→  const total = projects.length;
    70→  const ended = projects.filter((p) => p.status === "ENDED").length;
    71→  const running = projects.filter((p) => p.status === "RUNNING").length;
    72→  const pending = projects.filter((p) => p.status === "PENDING").length;
    73→  const paused = projects.filter((p) => p.status === "PAUSED").length;
    74→
    75→  // Weekly completion = tasks completed in last 7 days / total tasks
    76→  const weekAgo = new Date();
    77→  weekAgo.setDate(weekAgo.getDate() - 7);
    78→  const completedThisWeek = tasks.filter(
    79→    (t) => t.status === "DONE"
    80→  ).length;
    81→  const weeklyCompletion = tasks.length
    82→    ? Math.round((completedThisWeek / tasks.length) * 100)
    83→    : 0;
    84→
    85→  const projectEndedPct = total ? Math.round((ended / total) * 100) : 0;
    86→
    87→  return {
    88→    projects: projects.map(p => ({
    89→      ...p,
    90→      dueDate: p.dueDate?.toISOString() ?? null,
    91→      createdAt: p.createdAt.toISOString(),
    92→    })),
    93→    tasks: tasks.map(t => ({
    94→      ...t,
    95→      dueDate: t.dueDate?.toISOString() ?? null,
    96→      createdAt: t.createdAt.toISOString(),
    97→      project: t.project ? {
    98→        ...t.project,
    99→        dueDate: t.project.dueDate?.toISOString() ?? null,
   100→        createdAt: t.project.createdAt.toISOString(),
   101→      } : null,
   102→      assignee: t.assignee ? {
   103→        ...t.assignee,
   104→        createdAt: t.assignee.createdAt.toISOString(),
   105→        updatedAt: t.assignee.updatedAt.toISOString(),
   106→      } : null,
   107→    })),
   108→    members: members.map(m => ({
   109→      ...m,
   110→      createdAt: m.createdAt.toISOString(),
   111→      updatedAt: m.updatedAt.toISOString(),
   112→    })),
   113→    reminders: reminders.map(r => ({
   114→      ...r,
   115→      dueDate: r.dueDate.toISOString(),
   116→      createdAt: r.createdAt.toISOString(),
   117→    })),
   118→    notifications: notifs.map(n => ({
   119→      ...n,
   120→      createdAt: n.createdAt.toISOString(),
   121→    })),
   122→    timeEntries: timeEntries.map(te => ({
   123→      ...te,
   124→      date: te.date.toISOString(),
   125→      createdAt: te.createdAt.toISOString(),
   126→    })),
   127→    events: events.map(e => ({
   128→      ...e,
   129→      date: e.date.toISOString(),
   130→      createdAt: e.createdAt.toISOString(),
   131→    })),
   132→    messages: messages.map(m => ({
   133→      ...m,
   134→      createdAt: m.createdAt.toISOString(),
   135→    })),
   136→    stats: {
   137→      total, ended, running, pending, paused,
   138→      weeklyCompletion,
   139→      projectEndedPct,
   140→    },
   141→  };
   142→}

   162→export async function createTask(input: {
   163→  title: string;
   164→  description?: string;
   165→  status?: string;
   166→  priority?: string;
   167→  tag?: string;
   168→  dueDate?: string | null;
   169→  projectId?: string | null;
   170→  assigneeId?: string | null;
   171→}) {
   172→  const auth = await requirePermission("task:create");
   173→  const maxOrder = await db.task.aggregate({
   174→    where: { status: input.status ?? "TODO" },
   175→    _max: { order: true },
   176→  });
   177→  const task = await db.task.create({
   178→    data: {
   179→      title: input.title,
   180→      description: input.description ?? "",
   181→      status: input.status ?? "TODO",
   182→      priority: input.priority ?? "MEDIUM",
   183→      tag: input.tag ?? "FRONTEND",
   184→      dueDate: input.dueDate ? new Date(input.dueDate) : null,
   185→      projectId: input.projectId || null,
   186→      assigneeId: input.assigneeId || null,
   187→      order: (maxOrder._max.order ?? -1) + 1,
   188→    },
   189→    include: { project: true },
   190→  });
   191→  await recalcMemberStats(input.assigneeId);
   192→  await logAudit({
   193→    actor: auth,
   194→    action: "task.created",
   195→    entityType: "TASK",
   196→    entityId: task.id,
   197→    entityLabel: task.title,
   198→    detail: task.project
   199→      ? `Created task “${task.title}” in ${task.project.name}`
   200→      : `Created task “${task.title}” without a project`,
   201→  });
   202→  revalidatePath("/");
   203→}

   205→export async function updateTask(id: string, input: Partial<{
   206→  title: string;
   207→  description: string;
   208→  status: string;
   209→  priority: string;
   210→  tag: string;
   211→  dueDate: string | null;
   212→  projectId: string | null;
   213→  assigneeId: string | null;
   214→}>) {
   215→  // task:edit:any allows editing any task; otherwise check task:edit:own
   216→  const auth = await requirePermission("task:edit:own");
   217→  const existing = await db.task.findUnique({ where: { id } });
   218→  if (!existing) return;
   219→  await db.task.update({
   220→    where: { id },
   221→    data: {
   222→      title: input.title,
   223→      description: input.description,
   224→      status: input.status,
   225→      priority: input.priority,
   226→      tag: input.tag,
   227→      dueDate: input.dueDate !== undefined ? (input.dueDate ? new Date(input.dueDate) : null) : undefined,
   228→      projectId: input.projectId,
   229→      assigneeId: input.assigneeId,
   230→    },
   231→  });
   232→  await recalcMemberStats(existing.assigneeId);
   233→  if (input.assigneeId && input.assigneeId !== existing.assigneeId) {
   234→    await recalcMemberStats(input.assigneeId);
   235→  }
   236→  const changes: string[] = [];
   237→  if (input.title !== undefined && input.title !== existing.title) changes.push("title");
   238→  if (input.status !== undefined && input.status !== existing.status) changes.push("status");
   239→  if (input.priority !== undefined && input.priority !== existing.priority) changes.push("priority");
   240→  if (input.tag !== undefined && input.tag !== existing.tag) changes.push("tag");
   241→  if (input.description !== undefined && input.description !== existing.description) changes.push("description");
   242→  if (input.dueDate !== undefined) {
   243→    const nd = input.dueDate ? new Date(input.dueDate).toISOString() : null;
   244→    const od = existing.dueDate ? existing.dueDate.toISOString() : null;
   245→    if (nd !== od) changes.push("due date");
   246→  }
   247→  if (input.projectId !== undefined && input.projectId !== existing.projectId) changes.push("project");
   248→  if (input.assigneeId !== undefined && input.assigneeId !== existing.assigneeId) changes.push("assignee");
   249→  await logAudit({
   250→    actor: auth,
   251→    action: "task.updated",
   252→    entityType: "TASK",
   253→    entityId: id,
   254→    entityLabel: input.title ?? existing.title,
   255→    detail: changes.length
   256→      ? `Updated ${changes.join(", ")} on “${input.title ?? existing.title}”`
   257→      : `Saved task “${input.title ?? existing.title}”`,
   258→  });
   259→  revalidatePath("/");
   260→}

   262→export async function moveTask(id: string, newStatus: string, newOrder: number) {
   263→  const auth = await requirePermission("task:edit:own");
   264→  const task = await db.task.findUnique({ where: { id } });
   265→  if (!task) return;
   266→  const oldAssignee = task.assigneeId;
   267→  await db.task.update({
   268→    where: { id },
   269→    data: { status: newStatus, order: newOrder },
   270→  });
   271→  await recalcMemberStats(oldAssignee);
   272→  if (task.status !== newStatus) {
   273→    await logAudit({
   274→      actor: auth,
   275→      action: "task.moved",
   276→      entityType: "TASK",
   277→      entityId: id,
   278→      entityLabel: task.title,
   279→      detail: `Moved “${task.title}” to ${STATUS_LABELS[newStatus] ?? newStatus}`,
   280→    });
   281→  }
   282→  revalidatePath("/");
   283→}

   302→async function recalcMemberStats(memberId: string | null) {
   303→  if (!memberId) return;
   304→  const tasks = await db.task.findMany({ where: { assigneeId: memberId } });
   305→  const open = tasks.filter((t) => t.status !== "DONE").length;
   306→  const completed = tasks.filter((t) => t.status === "DONE").length;
   307→  const workload = Math.min(100, Math.round((open / 6) * 100));
   308→  await db.teamMember.update({
   309→    where: { id: memberId },
   310→    data: { openTasks: open, completedTasks: completed, workload },
   311→  });
   312→}

   574→export async function toggleReminder(id: string, completed: boolean) {
   575→  const auth = await getSession();
   576→  if (!auth) throw new AuthorizationError("You must be signed in.");
   577→  const existing = await db.reminder.findUnique({ where: { id } });
   578→  await db.reminder.update({ where: { id }, data: { completed } });
   579→  if (existing && existing.completed !== completed) {
   580→    await logAudit({
   581→      actor: auth,
   582→      action: completed ? "reminder.completed" : "reminder.reopened",
   583→      entityType: "REMINDER",
   584→      entityId: id,
   585→      entityLabel: existing.title,
   586→      detail: completed
   587→        ? `Completed reminder “${existing.title}”`
   588→        : `Reopened reminder “${existing.title}”`,
   589→    });
   590→  }
   591→  revalidatePath("/");

   613→export async function updateProfile(userId: string, input: {
   614→  name: string;
   615→  role: string;
   616→  timezone: string;
   617→  bio: string;
   618→  avatarColor: string;
   619→}) {
   620→  const auth = await getSession();
   621→  if (!auth) throw new AuthorizationError("You must be signed in.");
   622→  if (auth.userId !== userId) {
   623→    throw new AuthorizationError("You can only update your own profile.");
   624→  }
   625→  const existing = await db.user.findUnique({ where: { id: userId } });
   626→  await db.user.update({ where: { id: userId }, data: input });
   627→  if (existing) {
   628→    const changes: string[] = [];
   629→    if (input.name !== existing.name) changes.push("name");
   630→    if (input.role !== existing.role) changes.push("job title");
   631→    if (input.timezone !== existing.timezone) changes.push("timezone");
   632→    if (input.bio !== existing.bio) changes.push("bio");
   633→    if (input.avatarColor !== existing.avatarColor) changes.push("avatar color");
   634→    await logAudit({
   635→      actor: auth,
   636→      action: "profile.updated",
   637→      entityType: "USER",
   638→      entityId: userId,
   639→      entityLabel: input.name,
   640→      detail: changes.length
   641→        ? `Updated ${changes.join(", ")} on their profile`
   642→        : "Saved their profile",
   643→    });
   644→  }
   645→  revalidatePath("/");
   646→}

   688→export async function sendMessage(input: {
   689→  direction: "OUT" | "IN";
   690→  threadId?: string;
   691→  memberId?: string | null;
   692→  senderName: string;
   693→  senderAvatarColor?: string;
   694→  senderRole?: string;
   695→  recipientName: string;
   696→  subject: string;
   697→  body: string;
   698→}) {
   699→  const auth = await requirePermission("message:send");
   700→  // Generate a threadId if not provided (new conversation)
   701→  const threadId = input.threadId || `${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   702→  await db.message.create({
   703→    data: {
   704→      direction: input.direction,
   705→      threadId,
   706→      memberId: input.memberId || null,
   707→      senderName: input.senderName,
   708→      senderAvatarColor: input.senderAvatarColor || "forest",
   709→      senderRole: input.senderRole || "",
   710→      recipientName: input.recipientName,
   711→      subject: input.subject,
   712→      body: input.body,
   713→      read: input.direction === "OUT" ? true : false, // outgoing is always "read"
   714→    },
   715→  });
   716→  await logAudit({
   717→    actor: auth,
   718→    action: "message.sent",
   719→    entityType: "MESSAGE",
   720→    entityLabel: input.subject,
   721→    detail: `Sent “${input.subject}” to ${input.recipientName}`,
   722→  });
   723→  revalidatePath("/");
   724→  return threadId;
   725→}

   761→export async function updateUserRole(userId: string, newRole: string) {
   762→  const auth = await requirePermission("role:manage");
   763→  if (!["ADMIN", "MANAGER", "MEMBER"].includes(newRole)) {
   764→    throw new AuthorizationError("Invalid role. Must be ADMIN, MANAGER, or MEMBER.");
   765→  }
   766→  if (userId === auth.userId && newRole !== "ADMIN") {
   767→    throw new AuthorizationError("You cannot change your own role. Ask another admin to do it.");
   768→  }
   769→  const target = await db.user.findUnique({ where: { id: userId } });
   770→  if (!target) return;
   771→  await db.user.update({ where: { id: userId }, data: { role: newRole } });
   772→  const roleLabel = (r: string) => ROLE_LABELS[r as Role] ?? r;
   773→  await logAudit({
   774→    actor: auth,
   775→    action: "role.changed",
   776→    entityType: "ROLE",
   777→    entityId: userId,
   778→    entityLabel: target.name,
   779→    detail: `Changed role of ${target.name} from ${roleLabel(target.role)} to ${roleLabel(newRole)}`,
   780→  });
   781→  revalidatePath("/");
   782→}

   800→export async function createUserForMember(memberId: string) {

   816→      passwordHash,
   817→      role: "MEMBER",
   818→      timezone: "Pacific/Auckland",
   819→      bio: "",
   820→      avatarColor: member.avatarColor,
   821→    },
   822→  });
   823→  await logAudit({
