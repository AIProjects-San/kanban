    53→// ---------------------------------------------------------------------------
    54→// Dashboard data loader — single round trip
    55→// ---------------------------------------------------------------------------
    56→//
    57→// Access scope:
    58→//   • ADMIN / MANAGER — sees every project, task and member in the workspace.
    59→//   • MEMBER          — sees ONLY projects they own or have tasks assigned to,
    60→//                       and only tasks belonging to those projects (plus any
    61→//                       task directly assigned to them). Other members are
    62→//                       still listed so they can be messaged, but their
    63→//                       workload stats are not exposed.
    64→//
    65→// The caller passes the requesting user so we can apply the filter at the
    66→// database layer (no leaking of records the user should not see).
    67→
    68→export async function getDashboardData(opts?: {
    69→  userRole?: string;
    70→  userName?: string;
    71→}) {
    72→  const role = opts?.userRole;
    73→  const userName = opts?.userName ?? "";
    74→  const isMember = role === "MEMBER";
    75→
    76→  // For members, compute the set of project IDs they are associated with.
    77→  // Association = owns the project (ownerName === userName) OR is the
    78→  // assignee of at least one task in that project OR is explicitly assigned
    79→  // via the ProjectAssignment join table.
    80→  let memberProjectIds: Set<string> | null = null;
    81→  if (isMember && userName) {
    82→    // We need the user's id to look up explicit assignments — get it from the
    83→    // session (the caller already verified the session in page.tsx).
    84→    const session = await getSession();
    85→    const [owned, assignedProjects, explicitAssignments] = await Promise.all([
    86→      db.project.findMany({
    87→        where: { ownerName: userName },
    88→        select: { id: true },
    89→      }),
    90→      db.task.findMany({
    91→        where: { assignee: { name: userName } },
    92→        select: { projectId: true },
    93→        distinct: ["projectId"],
    94→      }),
    95→      session?.userId
    96→        ? db.projectAssignment.findMany({
    97→            where: { userId: session.userId },
    98→            select: { projectId: true },
    99→          })
   100→        : Promise.resolve([]),
   101→    ]);
   102→    memberProjectIds = new Set([
   103→      ...owned.map((p) => p.id),
   104→      ...assignedProjects.map((t) => t.projectId).filter(Boolean) as string[],
   105→      ...explicitAssignments.map((a) => a.projectId),
   106→    ]);
   107→  }
   108→
   109→  const projectWhere = isMember && memberProjectIds
   110→    ? { id: { in: Array.from(memberProjectIds) } }
   111→    : undefined;
   112→
   113→  const taskWhere = isMember && memberProjectIds
   114→    ? {
   115→        OR: [
   116→          { projectId: { in: Array.from(memberProjectIds) } },
   117→          { assignee: { name: userName } },
   118→        ],
   119→      }
   120→    : undefined;
   121→
   122→  const [projects, tasks, members, reminders, notifs, timeEntries, events, messages] = await Promise.all([
   123→    db.project.findMany({ where: projectWhere, orderBy: { createdAt: "desc" } }),
   124→    db.task.findMany({
   125→      where: taskWhere,
   126→      include: { project: true, assignee: true },
   127→      orderBy: { order: "asc" },
   128→    }),
   129→    db.teamMember.findMany({ orderBy: { name: "asc" } }),
   130→    db.reminder.findMany({ orderBy: { dueDate: "asc" } }),
   131→    db.notification.findMany({ orderBy: { createdAt: "desc" }, take: 20 }),
   132→    db.timeEntry.findMany({ orderBy: { date: "desc" } }),
   133→    db.calendarEvent.findMany({ orderBy: { date: "asc" } }),
   134→    db.message.findMany({ orderBy: { createdAt: "desc" } }),
   135→  ]);
   136→
   137→  const total = projects.length;
   138→  const ended = projects.filter((p) => p.status === "ENDED").length;
   139→  const running = projects.filter((p) => p.status === "RUNNING").length;
   140→  const pending = projects.filter((p) => p.status === "PENDING").length;
   141→  const paused = projects.filter((p) => p.status === "PAUSED").length;
   142→
   143→  // Weekly completion = tasks completed in last 7 days / total tasks
   144→  const weekAgo = new Date();
   145→  weekAgo.setDate(weekAgo.getDate() - 7);
   146→  const completedThisWeek = tasks.filter(
   147→    (t) => t.status === "DONE"
   148→  ).length;
   149→  const weeklyCompletion = tasks.length
   150→    ? Math.round((completedThisWeek / tasks.length) * 100)
   151→    : 0;
   152→
   153→  const projectEndedPct = total ? Math.round((ended / total) * 100) : 0;
   154→
   155→  return {
   156→    projects: projects.map(p => ({
   157→      ...p,
   158→      dueDate: p.dueDate?.toISOString() ?? null,
   159→      createdAt: p.createdAt.toISOString(),
   160→    })),
   161→    tasks: tasks.map(t => ({
   162→      ...t,
   163→      dueDate: t.dueDate?.toISOString() ?? null,
   164→      createdAt: t.createdAt.toISOString(),
   165→      project: t.project ? {
   166→        ...t.project,
   167→        dueDate: t.project.dueDate?.toISOString() ?? null,
   168→        createdAt: t.project.createdAt.toISOString(),
   169→      } : null,
   170→      assignee: t.assignee ? {
   171→        ...t.assignee,
   172→        createdAt: t.assignee.createdAt.toISOString(),
   173→        updatedAt: t.assignee.updatedAt.toISOString(),
   174→      } : null,
   175→    })),
   176→    members: members.map(m => ({
   177→      ...m,
   178→      createdAt: m.createdAt.toISOString(),
   179→      updatedAt: m.updatedAt.toISOString(),
   180→    })),
   181→    reminders: reminders.map(r => ({
   182→      ...r,
   183→      dueDate: r.dueDate.toISOString(),
   184→      createdAt: r.createdAt.toISOString(),
   185→    })),
   186→    notifications: notifs.map(n => ({
   187→      ...n,
   188→      createdAt: n.createdAt.toISOString(),
   189→    })),
   190→    timeEntries: timeEntries.map(te => ({
   191→      ...te,
   192→      date: te.date.toISOString(),
   193→      createdAt: te.createdAt.toISOString(),
   194→    })),
   195→    events: events.map(e => ({
   196→      ...e,
   197→      date: e.date.toISOString(),
   198→      createdAt: e.createdAt.toISOString(),
   199→    })),
   200→    messages: messages.map(m => ({
   201→      ...m,
   202→      createdAt: m.createdAt.toISOString(),
   203→    })),
   204→    stats: {
   205→      total, ended, running, pending, paused,
   206→      weeklyCompletion,
   207→      projectEndedPct,
   208→    },
   209→    // Surface the access scope so the client can show a banner / disable
   210→    // filters that don't make sense for a member.
   211→    scope: {
   212→      isMember: !!isMember,
   213→      projectIds: memberProjectIds ? Array.from(memberProjectIds) : null,
   214→    },
   215→  };
   216→}

   236→export async function createTask(input: {
   237→  title: string;
   238→  description?: string;
   239→  status?: string;
   240→  priority?: string;
   241→  tag?: string;
   242→  dueDate?: string | null;
   243→  projectId?: string | null;
   244→  assigneeId?: string | null;
   245→}) {
   246→  const auth = await requirePermission("task:create");
   247→  const maxOrder = await db.task.aggregate({
   248→    where: { status: input.status ?? "TODO" },
   249→    _max: { order: true },
   250→  });
   251→  const task = await db.task.create({
   252→    data: {
   253→      title: input.title,
   254→      description: input.description ?? "",
   255→      status: input.status ?? "TODO",
   256→      priority: input.priority ?? "MEDIUM",
   257→      tag: input.tag ?? "FRONTEND",
   258→      dueDate: input.dueDate ? new Date(input.dueDate) : null,
   259→      projectId: input.projectId || null,
   260→      assigneeId: input.assigneeId || null,
   261→      order: (maxOrder._max.order ?? -1) + 1,
   262→    },
   263→    include: { project: true },
   264→  });
   265→  await recalcMemberStats(input.assigneeId);
   266→  await logAudit({
   267→    actor: auth,
   268→    action: "task.created",
   269→    entityType: "TASK",
   270→    entityId: task.id,
   271→    entityLabel: task.title,
   272→    detail: task.project
   273→      ? `Created task “${task.title}” in ${task.project.name}`
   274→      : `Created task “${task.title}” without a project`,
   275→  });
   276→  revalidatePath("/");
   277→}

   279→export async function updateTask(id: string, input: Partial<{
   280→  title: string;
   281→  description: string;
   282→  status: string;
   283→  priority: string;
   284→  tag: string;
   285→  dueDate: string | null;
   286→  projectId: string | null;
   287→  assigneeId: string | null;
   288→}>) {
   289→  // task:edit:any allows editing any task; otherwise check task:edit:own
   290→  const auth = await requirePermission("task:edit:own");
   291→  const existing = await db.task.findUnique({ where: { id } });
   292→  if (!existing) return;
   293→  await db.task.update({
   294→    where: { id },
   295→    data: {
   296→      title: input.title,
   297→      description: input.description,
   298→      status: input.status,
   299→      priority: input.priority,
   300→      tag: input.tag,
   301→      dueDate: input.dueDate !== undefined ? (input.dueDate ? new Date(input.dueDate) : null) : undefined,
   302→      projectId: input.projectId,
   303→      assigneeId: input.assigneeId,
   304→    },
   305→  });
   306→  await recalcMemberStats(existing.assigneeId);
   307→  if (input.assigneeId && input.assigneeId !== existing.assigneeId) {
   308→    await recalcMemberStats(input.assigneeId);
   309→  }
   310→  const changes: string[] = [];
   311→  if (input.title !== undefined && input.title !== existing.title) changes.push("title");
   312→  if (input.status !== undefined && input.status !== existing.status) changes.push("status");
   313→  if (input.priority !== undefined && input.priority !== existing.priority) changes.push("priority");
   314→  if (input.tag !== undefined && input.tag !== existing.tag) changes.push("tag");
   315→  if (input.description !== undefined && input.description !== existing.description) changes.push("description");
   316→  if (input.dueDate !== undefined) {
   317→    const nd = input.dueDate ? new Date(input.dueDate).toISOString() : null;
   318→    const od = existing.dueDate ? existing.dueDate.toISOString() : null;
   319→    if (nd !== od) changes.push("due date");
   320→  }
   321→  if (input.projectId !== undefined && input.projectId !== existing.projectId) changes.push("project");
   322→  if (input.assigneeId !== undefined && input.assigneeId !== existing.assigneeId) changes.push("assignee");
   323→  await logAudit({
   324→    actor: auth,
   325→    action: "task.updated",
   326→    entityType: "TASK",
   327→    entityId: id,
   328→    entityLabel: input.title ?? existing.title,
   329→    detail: changes.length
   330→      ? `Updated ${changes.join(", ")} on “${input.title ?? existing.title}”`
   331→      : `Saved task “${input.title ?? existing.title}”`,
   332→  });
   333→  revalidatePath("/");
   334→}

   336→export async function moveTask(id: string, newStatus: string, newOrder: number) {
   337→  const auth = await requirePermission("task:edit:own");
   338→  const task = await db.task.findUnique({ where: { id } });
   339→  if (!task) return;
   340→  const oldAssignee = task.assigneeId;
   341→  await db.task.update({
   342→    where: { id },
   343→    data: { status: newStatus, order: newOrder },
   344→  });
   345→  await recalcMemberStats(oldAssignee);
   346→  if (task.status !== newStatus) {
   347→    await logAudit({
   348→      actor: auth,
   349→      action: "task.moved",
   350→      entityType: "TASK",
   351→      entityId: id,
   352→      entityLabel: task.title,
   353→      detail: `Moved “${task.title}” to ${STATUS_LABELS[newStatus] ?? newStatus}`,
   354→    });
   355→  }
   356→  revalidatePath("/");
   357→}

   368→    entityType: "TASK",
   369→    entityId: id,

   375→
   376→async function recalcMemberStats(memberId: string | null) {
   377→  if (!memberId) return;
   378→  const tasks = await db.task.findMany({ where: { assigneeId: memberId } });
   379→  const open = tasks.filter((t) => t.status !== "DONE").length;
   380→  const completed = tasks.filter((t) => t.status === "DONE").length;
   381→  const workload = Math.min(100, Math.round((open / 6) * 100));
   382→  await db.teamMember.update({
   383→    where: { id: memberId },
   384→    data: { openTasks: open, completedTasks: completed, workload },
   385→  });
   386→}

   848→// Full User CRUD — complete user management module
   849→// ---------------------------------------------------------------------------
   850→//
   851→// These actions go beyond `createUserForMember` (which only creates a login
   852→// for an existing team member). They let an admin create, edit, and delete
   853→// user accounts from scratch — with a custom email, password, and role — and
   854→// manage which projects each user is assigned to.
   855→//
   856→// When a new user is created we also create a matching TeamMember record so
   857→// the user appears in the team directory and can be assigned tasks.
   858→
   859→export async function createUser(input: {
   860→  name: string;
   861→  email: string;
   862→  password: string;
   863→  role: string;          // ADMIN | MANAGER | MEMBER
   864→  timezone?: string;
   865→  bio?: string;
   866→  avatarColor?: string;
   867→  department?: string;   // for the TeamMember record
   868→  jobTitle?: string;     // for the TeamMember record
   869→}) {
   870→  const auth = await requirePermission("role:manage");
   871→
   872→  // Validate
   873→  const name = input.name.trim();
   874→  const email = input.email.trim().toLowerCase();
   875→  if (!name) throw new AuthorizationError("Name is required.");
   876→  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   877→    throw new AuthorizationError("A valid email is required.");
   878→  }
   879→  if (!input.password || input.password.length < 8) {
   880→    throw new AuthorizationError("Password must be at least 8 characters.");
   881→  }
   882→  if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
   883→    throw new AuthorizationError("Invalid role. Must be ADMIN, MANAGER, or MEMBER.");
   884→  }
   885→
   886→  // Check email uniqueness
   887→  const existingUser = await db.user.findUnique({ where: { email } });
   888→  if (existingUser) {
   889→    throw new AuthorizationError(`A user with email ${email} already exists.`);
   890→  }
   891→
   892→  const bcrypt = (await import("bcryptjs")).default;
   893→  const passwordHash = await bcrypt.hash(input.password, 10);
   894→  const avatarColor = input.avatarColor ?? "forest";
   895→  const timezone = input.timezone ?? "Pacific/Auckland";
   896→  const bio = input.bio ?? "";
   897→  const department = input.department ?? "ENGINEERING";
   898→  const jobTitle = input.jobTitle ?? "Team Member";
   899→
   900→  // Create the User (login account) — single click, login created in the same
   901→  // transaction.
   902→  const user = await db.user.create({
   903→    data: {
   904→      email,
   905→      name,
   906→      passwordHash,
   907→      role: input.role,
   908→      timezone,
   909→      bio,
   910→      avatarColor,
   911→    },
   912→  });
   913→
   914→  // Also create a matching TeamMember record (if one doesn't already exist
   915→  // with the same email) so the user shows up in the team directory and can
   916→  // be assigned tasks.
   917→  const existingMember = await db.teamMember.findUnique({ where: { email } });
   918→  if (!existingMember) {
   919→    await db.teamMember.create({
   920→      data: {
   921→        name,
   922→        email,
   923→        role: jobTitle,
   924→        department,
   925→        avatarColor,
   926→        online: false,
   927→      },
   928→    });
   929→  }
   930→
   931→  await logAudit({
   932→    actor: auth,
   933→    action: "user.created",
   934→    entityType: "USER",
   935→    entityId: user.id,
   936→    entityLabel: name,
   937→    detail: `Created user ${name} (${email}) with role ${ROLE_LABELS[input.role as Role] ?? input.role}`,
   938→  });
   939→  revalidatePath("/");
   940→  return { id: user.id };
   941→}

   943→export async function updateUser(userId: string, input: {
   944→  name?: string;
   945→  email?: string;
   946→  role?: string;
   947→  timezone?: string;
   948→  bio?: string;
   949→  avatarColor?: string;
   950→  department?: string;
   951→  jobTitle?: string;
   952→  resetPassword?: string;  // if provided, set a new password
   953→}) {
   954→  const auth = await requirePermission("role:manage");
   955→  const target = await db.user.findUnique({ where: { id: userId } });
   956→  if (!target) throw new AuthorizationError("User not found.");
   957→
   958→  // Self-protection: cannot demote yourself
   959→  if (userId === auth.userId && input.role && input.role !== "ADMIN") {
   960→    throw new AuthorizationError("You cannot change your own role. Ask another admin to do it.");
   961→  }
   962→
   963→  const data: Record<string, unknown> = {};
   964→  if (input.name !== undefined) data.name = input.name.trim();
   965→  if (input.email !== undefined) {
   966→    const email = input.email.trim().toLowerCase();
   967→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   968→      throw new AuthorizationError("A valid email is required.");
   969→    }
   970→    data.email = email;
   971→  }
   972→  if (input.role !== undefined) {
   973→    if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
   974→      throw new AuthorizationError("Invalid role.");
   975→    }
   976→    data.role = input.role;
   977→  }
   978→  if (input.timezone !== undefined) data.timezone = input.timezone;
   979→  if (input.bio !== undefined) data.bio = input.bio;
   980→  if (input.avatarColor !== undefined) data.avatarColor = input.avatarColor;
   981→
   982→  if (input.resetPassword !== undefined && input.resetPassword.length > 0) {
   983→    if (input.resetPassword.length < 8) {
   984→      throw new AuthorizationError("Password must be at least 8 characters.");
   985→    }
   986→    const bcrypt = (await import("bcryptjs")).default;
   987→    data.passwordHash = await bcrypt.hash(input.resetPassword, 10);
   988→  }
   989→
   990→  await db.user.update({ where: { id: userId }, data });
   991→
   992→  // Sync name/email/avatarColor to the matching TeamMember record (if any)
   993→  const memberUpdate: Record<string, unknown> = {};
   994→  if (input.name !== undefined) memberUpdate.name = input.name.trim();
   995→  if (input.email !== undefined) memberUpdate.email = input.email.trim().toLowerCase();
   996→  if (input.avatarColor !== undefined) memberUpdate.avatarColor = input.avatarColor;
   997→  if (input.jobTitle !== undefined) memberUpdate.role = input.jobTitle;
   998→  if (input.department !== undefined) memberUpdate.department = input.department;
   999→  if (Object.keys(memberUpdate).length > 0) {
  1000→    await db.teamMember.updateMany({
  1001→      where: { email: target.email },
  1002→      data: memberUpdate,
  1003→    });
  1004→  }
  1005→
  1006→  const changes: string[] = [];
  1007→  if (input.name !== undefined && input.name !== target.name) changes.push("name");
  1008→  if (input.email !== undefined && input.email !== target.email) changes.push("email");
  1009→  if (input.role !== undefined && input.role !== target.role) changes.push("role");
  1010→  if (input.timezone !== undefined && input.timezone !== target.timezone) changes.push("timezone");
  1011→  if (input.bio !== undefined && input.bio !== target.bio) changes.push("bio");
  1012→  if (input.avatarColor !== undefined && input.avatarColor !== target.avatarColor) changes.push("avatar");
  1013→  if (input.resetPassword) changes.push("password");
  1014→
  1015→  await logAudit({
  1016→    actor: auth,
  1017→    action: "user.updated",
  1018→    entityType: "USER",
  1019→    entityId: userId,
  1020→    entityLabel: input.name ?? target.name,
  1021→    detail: changes.length
  1022→      ? `Updated ${changes.join(", ")} for ${input.name ?? target.name}`
  1023→      : `Saved user ${input.name ?? target.name}`,
  1024→  });
  1025→  revalidatePath("/");
  1026→}

  1154→
  1155→/**
  1156→ * Audit history for a single entity (e.g. one task, one project, one member).
  1157→ *
  1158→ * Unlike `getAuditLogs()` (which requires the `audit:view` permission and is
  1159→ * meant for the workspace-wide Audit Log page), this is callable by anyone
  1160→ * who can see the entity — i.e. anyone signed in. The reasoning: the audit
  1161→ * trail of a task you're editing is part of that task's context, and hiding
  1162→ * it would let a bad actor edit a task without other assignees ever seeing
  1163→ * who touched it. The action still respects the underlying RBAC layer
  1164→ * (you can't edit a task you can't edit), so surfacing its history is safe.
  1165→ *
  1166→ * Results are capped at the most recent 50 entries to keep the modal light.
  1167→ */
  1168→export async function getAuditLogsForEntity(entityType: string, entityId: string) {
  1169→  const auth = await getSession();
  1170→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1171→  if (!entityId) return [];
  1172→  const logs = await db.auditLog.findMany({
  1173→    where: { entityType, entityId },
  1174→    orderBy: { createdAt: "desc" },
  1175→    take: 50,
  1176→  });
  1177→  return logs.map((l) => ({
  1178→    ...l,
  1179→    createdAt: l.createdAt.toISOString(),
  1180→  }));
  1181→}

  1204→// ---------------------------------------------------------------------------
  1205→// Members allocated to a specific project.
  1206→//
  1207→// "Allocated" means the member is either:
  1208→//   • the project's owner (Project.ownerName === member.name), OR
  1209→//   • the assignee of at least one task in that project.
  1210→//
  1211→// Used by the task modal to restrict the assignee dropdown to only people who
  1212→// are actually working on the selected project. Admins still get the full team
  1213→// list when no project is selected, but once a project is picked the same
  1214→// restriction applies — you can't assign a task to someone outside the project.
  1215→// ---------------------------------------------------------------------------
  1216→
  1217→export async function getProjectMembers(projectId: string) {
  1218→  const auth = await getSession();
  1219→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1220→
  1221→  const project = await db.project.findUnique({
  1222→    where: { id: projectId },
  1223→    select: { id: true, name: true, ownerName: true },
  1224→  });
  1225→  if (!project) return [];
  1226→
  1227→  // Members who already have at least one task in this project.
  1228→  const assignedMembers = await db.teamMember.findMany({
  1229→    where: { tasks: { some: { projectId } } },
  1230→    orderBy: { name: "asc" },
  1231→  });
  1232→
  1233→  // Always include the project owner (if they're a team member).
  1234→  const ownerMember = project.ownerName
  1235→    ? await db.teamMember.findFirst({ where: { name: project.ownerName } })
  1236→    : null;
  1237→
  1238→  const seen = new Set<string>();
  1239→  const merged: Array<{
  1240→    id: string;
  1241→    name: string;
  1242→    email: string;
  1243→    role: string;
  1244→    department: string;
  1245→    avatarColor: string;
  1246→    online: boolean;
  1247→    openTasks: number;
  1248→    completedTasks: number;
  1249→    workload: number;
  1250→    isOwner: boolean;
  1251→  }> = [];
  1252→
  1253→  const push = (m: typeof assignedMembers[number], isOwner: boolean) => {
  1254→    if (seen.has(m.id)) return;
  1255→    seen.add(m.id);
  1256→    merged.push({
  1257→      id: m.id,
  1258→      name: m.name,
  1259→      email: m.email,
  1260→      role: m.role,
  1261→      department: m.department,
  1262→      avatarColor: m.avatarColor,
  1263→      online: m.online,
  1264→      openTasks: m.openTasks,
  1265→      completedTasks: m.completedTasks,
  1266→      workload: m.workload,
  1267→      isOwner,
  1268→    });
  1269→  };
  1270→
  1271→  if (ownerMember) push(ownerMember, true);
  1272→  for (const m of assignedMembers) push(m, m.name === project.ownerName);
  1273→
  1274→  return merged;
  1275→}

  1281→export async function changePassword(input: {
  1282→  userId: string;
  1283→  currentPassword: string;
  1284→  newPassword: string;
  1285→}) {
  1286→  const auth = await getSession();
  1287→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1288→  if (auth.userId !== input.userId) {
  1289→    throw new AuthorizationError("You can only change your own password.");
  1290→  }
  1291→  if (!input.newPassword || input.newPassword.length < 8) {
  1292→    throw new AuthorizationError("New password must be at least 8 characters.");
  1293→  }
  1294→  if (input.currentPassword === input.newPassword) {
  1295→    throw new AuthorizationError("New password must be different from your current password.");
  1296→  }
  1297→
  1298→  const user = await db.user.findUnique({ where: { id: input.userId } });
  1299→  if (!user) throw new AuthorizationError("Account not found.");
  1300→
  1301→  const bcrypt = (await import("bcryptjs")).default;
  1302→  const matches = await bcrypt.compare(input.currentPassword, user.passwordHash);
  1303→  if (!matches) {
  1304→    throw new AuthorizationError("Current password is incorrect.");
  1305→  }
  1306→
  1307→  const newHash = await bcrypt.hash(input.newPassword, 10);
  1308→  await db.user.update({
  1309→    where: { id: input.userId },
  1310→    data: { passwordHash: newHash },
  1311→  });
  1312→
  1313→  await logAudit({
  1314→    actor: auth,
  1315→    action: "auth.password_changed",
  1316→    entityType: "AUTH",
  1317→    entityId: input.userId,
  1318→    entityLabel: user.email,
  1319→    detail: `${user.name} changed their password`,
  1320→  });
  1321→  revalidatePath("/");
  1322→}
