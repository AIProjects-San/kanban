    53→// ---------------------------------------------------------------------------
    54→// Dashboard data loader — single round trip
    55→// ---------------------------------------------------------------------------
    56→//
    57→// Access scope:
    58→//   • ADMIN / MANAGER — sees every project, task and member in the workspace.
    59→//   • MEMBER          — sees ONLY projects they own or have tasks assigned to,
    60→//                       and only tasks belonging to those projects (plus any
    61→//                       task directly assigned to them). Other members are
    62→//                       still listed so they can be messaged, but their
    63→//                       workload stats are not exposed.
    64→//
    65→// The caller passes the requesting user so we can apply the filter at the
    66→// database layer (no leaking of records the user should not see).
    67→
    68→export async function getDashboardData(opts?: {
    69→  userRole?: string;
    70→  userName?: string;
    71→}) {
    72→  const role = opts?.userRole;
    73→  const userName = opts?.userName ?? "";
    74→  const isMember = role === "MEMBER";
    75→
    76→  // For members, compute the set of project IDs they are associated with.
    77→  // Association = owns the project (ownerName === userName) OR is the
    78→  // assignee of at least one task in that project OR is explicitly assigned
    79→  // via the ProjectAssignment join table.
    80→  let memberProjectIds: Set<string> | null = null;
    81→  if (isMember && userName) {
    82→    // We need the user's id to look up explicit assignments — get it from the
    83→    // session (the caller already verified the session in page.tsx).
    84→    const session = await getSession();
    85→    const [owned, assignedProjects, explicitAssignments] = await Promise.all([
    86→      db.project.findMany({
    87→        where: { ownerName: userName },
    88→        select: { id: true },
    89→      }),
    90→      db.task.findMany({
    91→        where: { assignee: { name: userName } },
    92→        select: { projectId: true },
    93→        distinct: ["projectId"],
    94→      }),
    95→      session?.userId
    96→        ? db.projectAssignment.findMany({
    97→            where: { userId: session.userId },
    98→            select: { projectId: true },
    99→          })
   100→        : Promise.resolve([]),
   101→    ]);
   102→    memberProjectIds = new Set([
   103→      ...owned.map((p) => p.id),
   104→      ...assignedProjects.map((t) => t.projectId).filter(Boolean) as string[],
   105→      ...explicitAssignments.map((a) => a.projectId),
   106→    ]);
   107→  }
   108→
   109→  const projectWhere = isMember && memberProjectIds
   110→    ? { id: { in: Array.from(memberProjectIds) } }
   111→    : undefined;
   112→
   113→  const taskWhere = isMember && memberProjectIds
   114→    ? {
   115→        OR: [
   116→          { projectId: { in: Array.from(memberProjectIds) } },
   117→          { assignee: { name: userName } },
   118→        ],
   119→      }
   120→    : undefined;
   121→
   122→  const [projects, tasks, members, reminders, notifs, timeEntries, events, messages] = await Promise.all([
   123→    db.project.findMany({ where: projectWhere, orderBy: { createdAt: "desc" } }),
   124→    db.task.findMany({
   125→      where: taskWhere,
   126→      include: { project: true, assignee: true },
   127→      orderBy: { order: "asc" },
   128→    }),
   129→    db.teamMember.findMany({ orderBy: { name: "asc" } }),
   130→    db.reminder.findMany({ orderBy: { dueDate: "asc" } }),
   131→    db.notification.findMany({ orderBy: { createdAt: "desc" }, take: 20 }),
   132→    db.timeEntry.findMany({ orderBy: { date: "desc" } }),
   133→    db.calendarEvent.findMany({ orderBy: { date: "asc" } }),
   134→    db.message.findMany({ orderBy: { createdAt: "desc" } }),
   135→  ]);
   136→
   137→  const total = projects.length;
   138→  const ended = projects.filter((p) => p.status === "ENDED").length;
   139→  const running = projects.filter((p) => p.status === "RUNNING").length;
   140→  const pending = projects.filter((p) => p.status === "PENDING").length;
   141→  const paused = projects.filter((p) => p.status === "PAUSED").length;
   142→
   143→  // Weekly completion = tasks completed in last 7 days / total tasks
   144→  const weekAgo = new Date();
   145→  weekAgo.setDate(weekAgo.getDate() - 7);
   146→  const completedThisWeek = tasks.filter(
   147→    (t) => t.status === "DONE"
   148→  ).length;
   149→  const weeklyCompletion = tasks.length
   150→    ? Math.round((completedThisWeek / tasks.length) * 100)
   151→    : 0;
   152→
   153→  const projectEndedPct = total ? Math.round((ended / total) * 100) : 0;
   154→
   155→  return {
   156→    projects: projects.map(p => ({
   157→      ...p,
   158→      dueDate: p.dueDate?.toISOString() ?? null,
   159→      createdAt: p.createdAt.toISOString(),
   160→    })),
   161→    tasks: tasks.map(t => ({
   162→      ...t,
   163→      dueDate: t.dueDate?.toISOString() ?? null,
   164→      createdAt: t.createdAt.toISOString(),
   165→      project: t.project ? {
   166→        ...t.project,
   167→        dueDate: t.project.dueDate?.toISOString() ?? null,
   168→        createdAt: t.project.createdAt.toISOString(),
   169→      } : null,
   170→      assignee: t.assignee ? {
   171→        ...t.assignee,
   172→        createdAt: t.assignee.createdAt.toISOString(),
   173→        updatedAt: t.assignee.updatedAt.toISOString(),
   174→      } : null,
   175→    })),
   176→    members: members.map(m => ({
   177→      ...m,
   178→      createdAt: m.createdAt.toISOString(),
   179→      updatedAt: m.updatedAt.toISOString(),
   180→    })),
   181→    reminders: reminders.map(r => ({
   182→      ...r,
   183→      dueDate: r.dueDate.toISOString(),
   184→      createdAt: r.createdAt.toISOString(),
   185→    })),
   186→    notifications: notifs.map(n => ({
   187→      ...n,
   188→      createdAt: n.createdAt.toISOString(),
   189→    })),
   190→    timeEntries: timeEntries.map(te => ({
   191→      ...te,
   192→      date: te.date.toISOString(),
   193→      createdAt: te.createdAt.toISOString(),
   194→    })),
   195→    events: events.map(e => ({
   196→      ...e,
   197→      date: e.date.toISOString(),
   198→      createdAt: e.createdAt.toISOString(),
   199→    })),
   200→    messages: messages.map(m => ({
   201→      ...m,
   202→      createdAt: m.createdAt.toISOString(),
   203→    })),
   204→    stats: {
   205→      total, ended, running, pending, paused,
   206→      weeklyCompletion,
   207→      projectEndedPct,
   208→    },
   209→    // Surface the access scope so the client can show a banner / disable
   210→    // filters that don't make sense for a member.
   211→    scope: {
   212→      isMember: !!isMember,
   213→      projectIds: memberProjectIds ? Array.from(memberProjectIds) : null,
   214→    },
   215→  };
   216→}

   236→export async function createTask(input: {
   237→  title: string;
   238→  description?: string;
   239→  status?: string;
   240→  priority?: string;
   241→  tag?: string;
   242→  dueDate?: string | null;
   243→  projectId?: string | null;
   244→  assigneeId?: string | null;
   245→}) {
   246→  const auth = await requirePermission("task:create");
   247→  const maxOrder = await db.task.aggregate({
   248→    where: { status: input.status ?? "TODO" },
   249→    _max: { order: true },
   250→  });
   251→  const task = await db.task.create({
   252→    data: {
   253→      title: input.title,
   254→      description: input.description ?? "",
   255→      status: input.status ?? "TODO",
   256→      priority: input.priority ?? "MEDIUM",
   257→      tag: input.tag ?? "FRONTEND",
   258→      dueDate: input.dueDate ? new Date(input.dueDate) : null,
   259→      projectId: input.projectId || null,
   260→      assigneeId: input.assigneeId || null,
   261→      order: (maxOrder._max.order ?? -1) + 1,
   262→    },
   263→    include: { project: true },
   264→  });
   265→  await recalcMemberStats(input.assigneeId);
   266→  await logAudit({
   267→    actor: auth,
   268→    action: "task.created",
   269→    entityType: "TASK",
   270→    entityId: task.id,
   271→    entityLabel: task.title,
   272→    detail: task.project
   273→      ? `Created task “${task.title}” in ${task.project.name}`
   274→      : `Created task “${task.title}” without a project`,
   275→  });
   276→  revalidatePath("/");
   277→}

   279→export async function updateTask(id: string, input: Partial<{
   280→  title: string;
   281→  description: string;
   282→  status: string;
   283→  priority: string;
   284→  tag: string;
   285→  dueDate: string | null;
   286→  projectId: string | null;
   287→  assigneeId: string | null;
   288→}>) {
   289→  // task:edit:any allows editing any task; otherwise check task:edit:own
   290→  const auth = await requirePermission("task:edit:own");
   291→  const existing = await db.task.findUnique({ where: { id } });
   292→  if (!existing) return;
   293→  await db.task.update({
   294→    where: { id },
   295→    data: {
   296→      title: input.title,
   297→      description: input.description,
   298→      status: input.status,
   299→      priority: input.priority,
   300→      tag: input.tag,
   301→      dueDate: input.dueDate !== undefined ? (input.dueDate ? new Date(input.dueDate) : null) : undefined,
   302→      projectId: input.projectId,
   303→      assigneeId: input.assigneeId,
   304→    },
   305→  });
   306→  await recalcMemberStats(existing.assigneeId);
   307→  if (input.assigneeId && input.assigneeId !== existing.assigneeId) {
   308→    await recalcMemberStats(input.assigneeId);
   309→  }
   310→  const changes: string[] = [];
   311→  if (input.title !== undefined && input.title !== existing.title) changes.push("title");
   312→  if (input.status !== undefined && input.status !== existing.status) changes.push("status");
   313→  if (input.priority !== undefined && input.priority !== existing.priority) changes.push("priority");
   314→  if (input.tag !== undefined && input.tag !== existing.tag) changes.push("tag");
   315→  if (input.description !== undefined && input.description !== existing.description) changes.push("description");
   316→  if (input.dueDate !== undefined) {
   317→    const nd = input.dueDate ? new Date(input.dueDate).toISOString() : null;
   318→    const od = existing.dueDate ? existing.dueDate.toISOString() : null;
   319→    if (nd !== od) changes.push("due date");
   320→  }
   321→  if (input.projectId !== undefined && input.projectId !== existing.projectId) changes.push("project");
   322→  if (input.assigneeId !== undefined && input.assigneeId !== existing.assigneeId) changes.push("assignee");
   323→  await logAudit({
   324→    actor: auth,
   325→    action: "task.updated",
   326→    entityType: "TASK",
   327→    entityId: id,
   328→    entityLabel: input.title ?? existing.title,
   329→    detail: changes.length
   330→      ? `Updated ${changes.join(", ")} on “${input.title ?? existing.title}”`
   331→      : `Saved task “${input.title ?? existing.title}”`,
   332→  });
   333→  revalidatePath("/");
   334→}

   336→export async function moveTask(id: string, newStatus: string, newOrder: number) {
   337→  const auth = await requirePermission("task:edit:own");
   338→  const task = await db.task.findUnique({ where: { id } });
   339→  if (!task) return;
   340→  const oldAssignee = task.assigneeId;
   341→  await db.task.update({
   342→    where: { id },
   343→    data: { status: newStatus, order: newOrder },
   344→  });
   345→  await recalcMemberStats(oldAssignee);
   346→  if (task.status !== newStatus) {
   347→    await logAudit({
   348→      actor: auth,
   349→      action: "task.moved",
   350→      entityType: "TASK",
   351→      entityId: id,
   352→      entityLabel: task.title,
   353→      detail: `Moved “${task.title}” to ${STATUS_LABELS[newStatus] ?? newStatus}`,
   354→    });
   355→  }
   356→  revalidatePath("/");
   357→}

   368→    entityType: "TASK",
   369→    entityId: id,

   375→
   376→async function recalcMemberStats(memberId: string | null) {
   377→  if (!memberId) return;
   378→  const tasks = await db.task.findMany({ where: { assigneeId: memberId } });
   379→  const open = tasks.filter((t) => t.status !== "DONE").length;
   380→  const completed = tasks.filter((t) => t.status === "DONE").length;
   381→  const workload = Math.min(100, Math.round((open / 6) * 100));
   382→  await db.teamMember.update({
   383→    where: { id: memberId },
   384→    data: { openTasks: open, completedTasks: completed, workload },
   385→  });
   386→}

   908→// ---------------------------------------------------------------------------
   909→// Full User CRUD — complete user management module
   910→// ---------------------------------------------------------------------------
   911→//
   912→// These actions go beyond `createUserForMember` (which only creates a login
   913→// for an existing team member). They let an admin create, edit, and delete
   914→// user accounts from scratch — with a custom email, password, and role — and
   915→// manage which projects each user is assigned to.
   916→//
   917→// When a new user is created we also create a matching TeamMember record so
   918→// the user appears in the team directory and can be assigned tasks.
   919→
   920→export async function createUser(input: {
   921→  name: string;
   922→  email: string;
   923→  password: string;
   924→  role: string;          // ADMIN | MANAGER | MEMBER
   925→  timezone?: string;
   926→  bio?: string;
   927→  avatarColor?: string;
   928→  department?: string;   // for the TeamMember record
   929→  jobTitle?: string;     // for the TeamMember record
   930→}) {
   931→  const auth = await requirePermission("role:manage");
   932→
   933→  // Validate
   934→  const name = input.name.trim();
   935→  const email = input.email.trim().toLowerCase();
   936→  if (!name) throw new AuthorizationError("Name is required.");
   937→  if (!email || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
   938→    throw new AuthorizationError("A valid email is required.");
   939→  }
   940→  if (!input.password || input.password.length < 8) {
   941→    throw new AuthorizationError("Password must be at least 8 characters.");
   942→  }
   943→  if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
   944→    throw new AuthorizationError("Invalid role. Must be ADMIN, MANAGER, or MEMBER.");
   945→  }
   946→
   947→  // Check email uniqueness
   948→  const existingUser = await db.user.findUnique({ where: { email } });
   949→  if (existingUser) {
   950→    throw new AuthorizationError(`A user with email ${email} already exists.`);
   951→  }
   952→
   953→  const bcrypt = (await import("bcryptjs")).default;
   954→  const passwordHash = await bcrypt.hash(input.password, 10);
   955→  const avatarColor = input.avatarColor ?? "forest";
   956→  const timezone = input.timezone ?? "Pacific/Auckland";
   957→  const bio = input.bio ?? "";
   958→  const department = input.department ?? "ENGINEERING";
   959→  const jobTitle = input.jobTitle ?? "Team Member";
   960→
   961→  // Create the User (login account) — single click, login created in the same
   962→  // transaction.
   963→  const user = await db.user.create({
   964→    data: {
   965→      email,
   966→      name,
   967→      passwordHash,
   968→      role: input.role,
   969→      timezone,
   970→      bio,
   971→      avatarColor,
   972→    },
   973→  });
   974→
   975→  // Also create a matching TeamMember record (if one doesn't already exist
   976→  // with the same email) so the user shows up in the team directory and can
   977→  // be assigned tasks.
   978→  const existingMember = await db.teamMember.findUnique({ where: { email } });
   979→  if (!existingMember) {
   980→    await db.teamMember.create({
   981→      data: {
   982→        name,
   983→        email,
   984→        role: jobTitle,
   985→        department,
   986→        avatarColor,
   987→        online: false,
   988→      },
   989→    });
   990→  }
   991→
   992→  await logAudit({
   993→    actor: auth,
   994→    action: "user.created",
   995→    entityType: "USER",
   996→    entityId: user.id,
   997→    entityLabel: name,
   998→    detail: `Created user ${name} (${email}) with role ${ROLE_LABELS[input.role as Role] ?? input.role}`,
   999→  });
  1000→  revalidatePath("/");
  1001→  return { id: user.id };
  1002→}

  1004→export async function updateUser(userId: string, input: {
  1005→  name?: string;
  1006→  email?: string;
  1007→  role?: string;
  1008→  timezone?: string;
  1009→  bio?: string;
  1010→  avatarColor?: string;
  1011→  department?: string;
  1012→  jobTitle?: string;
  1013→  resetPassword?: string;  // if provided, set a new password
  1014→}) {
  1015→  const auth = await requirePermission("role:manage");
  1016→  const target = await db.user.findUnique({ where: { id: userId } });
  1017→  if (!target) throw new AuthorizationError("User not found.");
  1018→
  1019→  // Self-protection: cannot demote yourself
  1020→  if (userId === auth.userId && input.role && input.role !== "ADMIN") {
  1021→    throw new AuthorizationError("You cannot change your own role. Ask another admin to do it.");
  1022→  }
  1023→
  1024→  const data: Record<string, unknown> = {};
  1025→  if (input.name !== undefined) data.name = input.name.trim();
  1026→  if (input.email !== undefined) {
  1027→    const email = input.email.trim().toLowerCase();
  1028→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
  1029→      throw new AuthorizationError("A valid email is required.");
  1030→    }
  1031→    data.email = email;
  1032→  }
  1033→  if (input.role !== undefined) {
  1034→    if (!["ADMIN", "MANAGER", "MEMBER"].includes(input.role)) {
  1035→      throw new AuthorizationError("Invalid role.");
  1036→    }
  1037→    data.role = input.role;
  1038→  }
  1039→  if (input.timezone !== undefined) data.timezone = input.timezone;
  1040→  if (input.bio !== undefined) data.bio = input.bio;
  1041→  if (input.avatarColor !== undefined) data.avatarColor = input.avatarColor;
  1042→
  1043→  if (input.resetPassword !== undefined && input.resetPassword.length > 0) {
  1044→    if (input.resetPassword.length < 8) {
  1045→      throw new AuthorizationError("Password must be at least 8 characters.");
  1046→    }
  1047→    const bcrypt = (await import("bcryptjs")).default;
  1048→    data.passwordHash = await bcrypt.hash(input.resetPassword, 10);
  1049→  }
  1050→
  1051→  await db.user.update({ where: { id: userId }, data });
  1052→
  1053→  // Sync name/email/avatarColor to the matching TeamMember record (if any)
  1054→  const memberUpdate: Record<string, unknown> = {};
  1055→  if (input.name !== undefined) memberUpdate.name = input.name.trim();
  1056→  if (input.email !== undefined) memberUpdate.email = input.email.trim().toLowerCase();
  1057→  if (input.avatarColor !== undefined) memberUpdate.avatarColor = input.avatarColor;
  1058→  if (input.jobTitle !== undefined) memberUpdate.role = input.jobTitle;
  1059→  if (input.department !== undefined) memberUpdate.department = input.department;
  1060→  if (Object.keys(memberUpdate).length > 0) {
  1061→    await db.teamMember.updateMany({
  1062→      where: { email: target.email },
  1063→      data: memberUpdate,
  1064→    });
  1065→  }
  1066→
  1067→  const changes: string[] = [];
  1068→  if (input.name !== undefined && input.name !== target.name) changes.push("name");
  1069→  if (input.email !== undefined && input.email !== target.email) changes.push("email");
  1070→  if (input.role !== undefined && input.role !== target.role) changes.push("role");
  1071→  if (input.timezone !== undefined && input.timezone !== target.timezone) changes.push("timezone");
  1072→  if (input.bio !== undefined && input.bio !== target.bio) changes.push("bio");
  1073→  if (input.avatarColor !== undefined && input.avatarColor !== target.avatarColor) changes.push("avatar");
  1074→  if (input.resetPassword) changes.push("password");
  1075→
  1076→  await logAudit({
  1077→    actor: auth,
  1078→    action: "user.updated",
  1079→    entityType: "USER",
  1080→    entityId: userId,
  1081→    entityLabel: input.name ?? target.name,
  1082→    detail: changes.length
  1083→      ? `Updated ${changes.join(", ")} for ${input.name ?? target.name}`
  1084→      : `Saved user ${input.name ?? target.name}`,
  1085→  });
  1086→  revalidatePath("/");
  1087→}

  1216→/**
  1217→ * Audit history for a single entity (e.g. one task, one project, one member).
  1218→ *
  1219→ * Unlike `getAuditLogs()` (which requires the `audit:view` permission and is
  1220→ * meant for the workspace-wide Audit Log page), this is callable by anyone
  1221→ * who can see the entity — i.e. anyone signed in. The reasoning: the audit
  1222→ * trail of a task you're editing is part of that task's context, and hiding
  1223→ * it would let a bad actor edit a task without other assignees ever seeing
  1224→ * who touched it. The action still respects the underlying RBAC layer
  1225→ * (you can't edit a task you can't edit), so surfacing its history is safe.
  1226→ *
  1227→ * Results are capped at the most recent 50 entries to keep the modal light.
  1228→ */
  1229→export async function getAuditLogsForEntity(entityType: string, entityId: string) {
  1230→  const auth = await getSession();
  1231→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1232→  if (!entityId) return [];
  1233→  const logs = await db.auditLog.findMany({
  1234→    where: { entityType, entityId },
  1235→    orderBy: { createdAt: "desc" },
  1236→    take: 50,
  1237→  });
  1238→  return logs.map((l) => ({
  1239→    ...l,
  1240→    createdAt: l.createdAt.toISOString(),
  1241→  }));
  1242→}

  1265→// ---------------------------------------------------------------------------
  1266→// Members allocated to a specific project.
  1267→//
  1268→// "Allocated" means the member is either:
  1269→//   • the project's owner (Project.ownerName === member.name), OR
  1270→//   • the assignee of at least one task in that project.
  1271→//
  1272→// Used by the task modal to restrict the assignee dropdown to only people who
  1273→// are actually working on the selected project. Admins still get the full team
  1274→// list when no project is selected, but once a project is picked the same
  1275→// restriction applies — you can't assign a task to someone outside the project.
  1276→// ---------------------------------------------------------------------------
  1277→
  1278→export async function getProjectMembers(projectId: string) {
  1279→  const auth = await getSession();
  1280→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1281→
  1282→  const project = await db.project.findUnique({
  1283→    where: { id: projectId },
  1284→    select: { id: true, name: true, ownerName: true },
  1285→  });
  1286→  if (!project) return [];
  1287→
  1288→  // Members who already have at least one task in this project.
  1289→  const assignedMembers = await db.teamMember.findMany({
  1290→    where: { tasks: { some: { projectId } } },
  1291→    orderBy: { name: "asc" },
  1292→  });
  1293→
  1294→  // Always include the project owner (if they're a team member).
  1295→  const ownerMember = project.ownerName
  1296→    ? await db.teamMember.findFirst({ where: { name: project.ownerName } })
  1297→    : null;
  1298→
  1299→  const seen = new Set<string>();
  1300→  const merged: Array<{
  1301→    id: string;
  1302→    name: string;
  1303→    email: string;
  1304→    role: string;
  1305→    department: string;
  1306→    avatarColor: string;
  1307→    online: boolean;
  1308→    openTasks: number;
  1309→    completedTasks: number;
  1310→    workload: number;
  1311→    isOwner: boolean;
  1312→  }> = [];
  1313→
  1314→  const push = (m: typeof assignedMembers[number], isOwner: boolean) => {
  1315→    if (seen.has(m.id)) return;
  1316→    seen.add(m.id);
  1317→    merged.push({
  1318→      id: m.id,
  1319→      name: m.name,
  1320→      email: m.email,
  1321→      role: m.role,
  1322→      department: m.department,
  1323→      avatarColor: m.avatarColor,
  1324→      online: m.online,
  1325→      openTasks: m.openTasks,
  1326→      completedTasks: m.completedTasks,
  1327→      workload: m.workload,
  1328→      isOwner,
  1329→    });
  1330→  };
  1331→
  1332→  if (ownerMember) push(ownerMember, true);
  1333→  for (const m of assignedMembers) push(m, m.name === project.ownerName);
  1334→
  1335→  return merged;
  1336→}

  1342→export async function changePassword(input: {
  1343→  userId: string;
  1344→  currentPassword: string;
  1345→  newPassword: string;
  1346→}) {
  1347→  const auth = await getSession();
  1348→  if (!auth) throw new AuthorizationError("You must be signed in.");
  1349→  if (auth.userId !== input.userId) {
  1350→    throw new AuthorizationError("You can only change your own password.");
  1351→  }
  1352→  if (!input.newPassword || input.newPassword.length < 8) {
  1353→    throw new AuthorizationError("New password must be at least 8 characters.");
  1354→  }
  1355→  if (input.currentPassword === input.newPassword) {
  1356→    throw new AuthorizationError("New password must be different from your current password.");
  1357→  }
  1358→
  1359→  const user = await db.user.findUnique({ where: { id: input.userId } });
  1360→  if (!user) throw new AuthorizationError("Account not found.");
  1361→
  1362→  const bcrypt = (await import("bcryptjs")).default;
  1363→  const matches = await bcrypt.compare(input.currentPassword, user.passwordHash);
  1364→  if (!matches) {
  1365→    throw new AuthorizationError("Current password is incorrect.");
  1366→  }
  1367→
  1368→  const newHash = await bcrypt.hash(input.newPassword, 10);
  1369→  await db.user.update({
  1370→    where: { id: input.userId },
  1371→    data: { passwordHash: newHash },
  1372→  });
  1373→
  1374→  await logAudit({
  1375→    actor: auth,
  1376→    action: "auth.password_changed",
  1377→    entityType: "AUTH",
  1378→    entityId: input.userId,
  1379→    entityLabel: user.email,
  1380→    detail: `${user.name} changed their password`,
  1381→  });
  1382→  revalidatePath("/");
  1383→}
